   318→// ─────────────────────────────────────────────
   319→// Transform backend result to page format
   320→// ─────────────────────────────────────────────
   321→
   322→function transformBackendResult(result: any, initialCapital: number = 100000): BacktestResult {
   323→  if (!result) {
   324→    return {
   325→      metrics: {},
   326→      equityCurve: [],
   327→      drawdownChart: [],
   328→      monthlyReturns: [],
   329→      trades: [],
   330→      riskGateStats: [],
   331→    };
   332→  }
   333→
   334→  const r = result;
   335→  const capital = r.initial_capital || initialCapital;
   336→  const totalPnl = r.total_pnl ?? 0;
   337→  const totalReturn = capital > 0 ? parseFloat(((totalPnl / capital) * 100).toFixed(2)) : 0;
   338→  const totalTrades = r.total_trades ?? 0;
   339→  const winRate = r.win_rate ?? 0;
   340→  const avgWin = r.avg_win ?? 0;
   341→  const avgLoss = r.avg_loss ?? 0;
   342→  const avgTradePnl = totalTrades > 0 ? parseFloat((totalPnl / totalTrades).toFixed(2)) : 0;
   343→
   344→  // Equity curve
   345→  const equityCurve: EquityPoint[] = (r.equity_curve || []).map((pt: any, i: number) => ({
   346→    trade: i,
   347→    equity: Math.round(pt.capital ?? pt.equity ?? capital),
   348→  }));
   349→
   350→  // Drawdown chart — derive from equity curve
   351→  const drawdownChart: DrawdownPoint[] = [];
   352→  let peak = capital;
   353→  for (let i = 0; i < equityCurve.length; i++) {
   354→    const eq = equityCurve[i].equity;
   355→    peak = Math.max(peak, eq);
   356→    const dd = peak > 0 ? parseFloat((((eq - peak) / peak) * 100).toFixed(2)) : 0;
   357→    drawdownChart.push({
   358→      date: r.equity_curve?.[i]?.bar || `Day ${i}`,
   359→      drawdown: dd,
   360→    });
   361→  }
   362→
   363→  return {
   364→    metrics: {
   365→      totalReturn,
   366→      annualizedReturn: totalReturn, // Backend doesn't provide separate annualized
   367→      maxDrawdown: parseFloat((r.max_drawdown_pct ?? 0).toFixed(2)),
   368→      sharpe: parseFloat((r.sharpe_ratio ?? 0).toFixed(2)),
   369→      sortino: 0, // Backend doesn't provide
   370→      winRate: parseFloat((winRate * 100).toFixed(2)) || 0,
   371→      profitFactor: parseFloat((r.profit_factor ?? 0).toFixed(2)),
   372→      totalTrades,
   373→      totalFees: 0, // Backend doesn't provide
   374→      avgTradePnl: Math.round(avgTradePnl),
   375→    },
   376→    equityCurve,
   377→    drawdownChart,
   378→    monthlyReturns: [], // Backend doesn't provide monthly breakdown
   379→    trades: (r.trades || []).map((t: any, i: number) => ({
   380→      id: i + 1,
   381→      date: t.entry_time || t.date || 'N/A',
   382→      symbol: t.symbol || 'N/A',
   383→      direction: t.direction?.toUpperCase() || 'BUY',
   384→      entry: parseFloat(t.entry ?? 0),
   385→      exit: parseFloat(t.exit ?? 0),
   386→      pnl: parseFloat(t.pnl ?? 0),
   387→      duration: t.duration || '-',
   388→      exitReason: t.exit_reason || t.reason || '-',
   389→    })),
   390→    riskGateStats: [], // Backend doesn't provide risk gate breakdown
   391→  };
   392→}
