   770→   770→class RiskEngine:
   771→   771→    """
   772→   772→    13-gate risk validation engine.
   773→   773→    Every signal must pass ALL gates before being presented as opportunity.
   774→   774→    """
   775→   775→    
   776→   776→    def __init__(self, config, repository, daily_risk_manager):
   777→   777→        self.gates = [
   778→   778→            G1MaxPositions(config),
   779→   779→            G2SectorConcentration(config),
   780→   780→            G3MaxPositionSize(config),
   781→   781→            G4MaxDailyTrades(config),
   782→   782→            G5MaxDailyLoss(config),
   783→   783→            G6CorrelationCheck(config),
   784→   784→            G7VIXFilter(config),
   785→   785→            G8TimeOfDay(config),
   786→   786→            G9PriceMismatch(config),
   787→   787→            G10MinConfidence(config),
   788→   788→            G11MaxDrawdown(config),
   789→   789→            G12MarginCheck(config),
   790→   790→            G13DuplicateSignal(config),
   791→   791→        ]
   792→   792→        self.daily_risk = daily_risk_manager
   793→   793→    
   794→   794→    async def validate(self, signal, context) -> RiskResult:
   795→   795→        """
   796→   796→        Run signal through all 13 gates.
   797→   797→        Returns RiskResult with:
   798→   798→          - passed: bool
   799→   799→          - gate_results: list of GateResult (gate_name, passed, reason, details)
   800→   800→          - rejection: None or GateResult of first failing gate
   801→   801→        """
   802→   802→        results = []
   803→   803→        for gate in self.gates:
   804→   804→            result = await gate.check(signal, context)
   805→   805→            results.append(result)
   806→   806→            if not result.passed:
   807→   807→                # Log risk event
   808→   808→                await self.daily_risk.record_gate_rejection(gate, result, signal)
   809→   809→                return RiskResult(passed=False, gate_results=results, rejection=result)
   810→   810→        return RiskResult(passed=True, gate_results=results, rejection=None)
   811→   811→```
