    73→   207→    async def get_todays_pnl(self) -> Dict[str, Any]:
    74→   208→        """Get today's aggregate P&L."""
    75→   209→        today = _today_str()
    76→   210→        trades = await self.get_trades_by_date(today, limit=500)
    77→   211→        closed = [t for t in trades if t.status == "CLOSED"]
    78→   212→        gross_pnl = sum(t.pnl for t in closed)
    79→   213→        total_fees = sum(t.fees for t in closed) + sum(t.brokerage for t in closed)
    80→   214→        net_pnl = sum(t.net_pnl for t in closed)
    81→   215→        wins = sum(1 for t in closed if t.net_pnl > 0)
    82→   216→        losses = sum(1 for t in closed if t.net_pnl < 0)
    83→   217→        total = len(closed)
    84→   218→        return {
    85→   219→            "date": today,
    86→   220→            "total_trades": total,
    87→   221→            "wins": wins,
    88→   222→            "losses": losses,
    89→   223→            "win_rate": (wins / total * 100) if total > 0 else 0.0,
    90→   224→            "gross_pnl": round(gross_pnl, 2),
    91→   225→            "total_fees": round(total_fees, 2),
    92→   226→            "net_pnl": round(net_pnl, 2),
    93→   227→            "best_trade": round(max((t.net_pnl for t in closed), default=0), 2),
    94→   228→            "worst_trade": round(min((t.net_pnl for t in closed), default=0), 2),
    95→   229→        }

   627→   761→    async def get_max_drawdown_pct(self) -> float:
   628→   762→        """Calculate max drawdown percentage from all closed trades."""
   629→   763→        stmt = select(Trade).where(Trade.status == "CLOSED").order_by(Trade.exit_time.asc())
   630→   764→        result = await self.session.execute(stmt)
   631→   765→        trades = list(result.scalars().all())
   632→   766→        if not trades:
   633→   767→            return 0.0
   634→   768→        peak = 100000.0  # Starting capital
   635→   769→        max_dd = 0.0
   636→   770→        running = peak
   637→   771→        for t in trades:
   638→   772→            running += t.net_pnl
   639→   773→            if running > peak:
   640→   774→                peak = running
   641→   775→            dd = (peak - running) / peak * 100 if peak > 0 else 0.0
   642→   776→            if dd > max_dd:
   643→   777→                max_dd = dd
   644→   778→        return round(max_dd, 2)
