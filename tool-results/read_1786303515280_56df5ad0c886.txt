    76→        stmt = select(model).order_by(model.created_at.desc()).limit(limit).offset(offset)
    77→        result = await self.session.execute(stmt)
    78→        return list(result.scalars().all())

    83→        await self.session.flush()

   207→    async def get_todays_pnl(self) -> Dict[str, Any]:
   208→        """Get today's aggregate P&L."""
   209→        today = _today_str()
   210→        trades = await self.get_trades_by_date(today, limit=500)
   211→        closed = [t for t in trades if t.status == "CLOSED"]
   212→        gross_pnl = sum(t.pnl for t in closed)
   213→        total_fees = sum(t.fees for t in closed) + sum(t.brokerage for t in closed)
   214→        net_pnl = sum(t.net_pnl for t in closed)
   215→        wins = sum(1 for t in closed if t.net_pnl > 0)
   216→        losses = sum(1 for t in closed if t.net_pnl < 0)
   217→        total = len(closed)
   218→        return {
   219→            "date": today,
   220→            "total_trades": total,
   221→            "wins": wins,
   222→            "losses": losses,
   223→            "win_rate": (wins / total * 100) if total > 0 else 0.0,
   224→            "gross_pnl": round(gross_pnl, 2),
   225→            "total_fees": round(total_fees, 2),
   226→            "net_pnl": round(net_pnl, 2),
   227→            "best_trade": round(max((t.net_pnl for t in closed), default=0), 2),
   228→            "worst_trade": round(min((t.net_pnl for t in closed), default=0), 2),
   229→        }

   761→    async def get_max_drawdown_pct(self) -> float:
   762→        """Calculate max drawdown percentage from all closed trades."""
   763→        stmt = select(Trade).where(Trade.status == "CLOSED").order_by(Trade.exit_time.asc())
   764→        result = await self.session.execute(stmt)
   765→        trades = list(result.scalars().all())
   766→        if not trades:
   767→            return 0.0
   768→        peak = 100000.0  # Starting capital
   769→        max_dd = 0.0
   770→        running = peak
   771→        for t in trades:
   772→            running += t.net_pnl
   773→            if running > peak:
   774→                peak = running
   775→            dd = (peak - running) / peak * 100 if peak > 0 else 0.0
   776→            if dd > max_dd:
   777→                max_dd = dd
   778→        return round(max_dd, 2)
