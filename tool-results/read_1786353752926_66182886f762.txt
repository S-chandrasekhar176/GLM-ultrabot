   175→// ─────────────────────────────────────────────
   176→// Indian Number Formatting
   177→// ─────────────────────────────────────────────
   178→
   179→function formatINR(value: number): string {
   180→  const sign = value < 0 ? '-' : '';
   181→  const abs = Math.abs(value);
   182→  const parts = abs.toFixed(2).split('.');
   183→  const intPart = parts[0];
   184→  const decPart = parts[1];
   185→  let formatted = '';
   186→  if (intPart.length <= 3) {
   187→    formatted = intPart;
   188→  } else {
   189→    const last3 = intPart.slice(-3);
   190→    const rest = intPart.slice(0, -3);
   191→    formatted = rest.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',' + last3;
   192→  }
   193→  return `₹${sign}${formatted}.${decPart}`;
   194→}
   195→
   196→function formatPercent(value: number): string {
   197→  const sign = value >= 0 ? '+' : '';
   198→  return `${sign}${value.toFixed(2)}%`;
   199→}

   201→// ─────────────────────────────────────────────
   202→// Circular Progress (SVG)
   203→// ─────────────────────────────────────────────
   204→
   205→function CircularProgress({ value, size = 64, strokeWidth = 5, color = '#00d09c' }: {
   206→  value: number;
   207→  size?: number;
   208→  strokeWidth?: number;
   209→  color?: string;
   210→}) {
   211→  const radius = (size - strokeWidth) / 2;
   212→  const circumference = 2 * Math.PI * radius;
   213→  const offset = circumference - (value / 100) * circumference;
   214→
   215→  return (
   216→    <svg width={size} height={size} className="-rotate-90">
   217→      <circle
   218→        cx={size / 2}
   219→        cy={size / 2}
   220→        r={radius}
   221→        fill="none"
   222→        stroke="#1e293b"
   223→        strokeWidth={strokeWidth}
   224→      />
   225→      <circle
   226→        cx={size / 2}
   227→        cy={size / 2}
   228→        r={radius}
   229→        fill="none"
   230→        stroke={color}
   231→        strokeWidth={strokeWidth}
   232→        strokeDasharray={circumference}
   233→        strokeDashoffset={offset}
   234→        strokeLinecap="round"
   235→        className="transition-all duration-700 ease-out"
   236→      />
   237→      <text
   238→        x={size / 2}
   239→        y={size / 2}
   240→        className="fill-ub-text-primary text-xs font-bold"
   241→        textAnchor="middle"
   242→        dominantBaseline="central"
   243→        transform={`rotate(90, ${size / 2}, ${size / 2})`}
   244→      >
   245→        {value.toFixed(0)}%
   246→      </text>
   247→    </svg>
   248→  );
   249→}

   356→
   357→function MarketTimer() {
   358→  const [now, setNow] = useState(() => new Date());
   359→  const engine = useEngineStore();
   360→
   361→  useEffect(() => {
   362→    const interval = setInterval(() => setNow(new Date()), 1_000);
   363→    return () => clearInterval(interval);
   364→  }, []);
   365→
   366→  const istString = now.toLocaleTimeString('en-IN', {
   367→    timeZone: 'Asia/Kolkata',
   368→    hour12: false,
   369→    hour: '2-digit',
   370→    minute: '2-digit',
   371→    second: '2-digit',
   372→  });
   373→
   374→  const marketCloseSeconds = engine.marketCloseSeconds;
   375→  const hours = Math.floor(marketCloseSeconds / 3600);
   376→  const minutes = Math.floor((marketCloseSeconds % 3600) / 60);
   377→  const seconds = Math.floor(marketCloseSeconds % 60);
   378→  const pad = (n: number) => String(n).padStart(2, '0');
   379→  const timeToClose = marketCloseSeconds > 0
   380→    ? `${pad(hours)}:${pad(minutes)}:${pad(seconds)}`
   381→    : 'Market Closed';
   382→
   383→  const isUrgent = marketCloseSeconds > 0 && marketCloseSeconds < 1800;
   384→
   385→  return (
   386→    <div className="flex flex-col gap-2">
   387→      <div className="flex items-center gap-2">
   388→        <Clock className="h-4 w-4 text-ub-text-muted" />
   389→        <span className="text-lg font-mono font-bold text-ub-text-primary tracking-wider">
   390→          {istString}
   391→        </span>
   392→        <Badge variant="outline" className="text-[10px] font-medium border-ub-border text-ub-text-muted">
   393→          IST
   394→        </Badge>
   395→      </div>
   396→      <Separator className="bg-ub-border" />
   397→      <div className="flex items-center justify-between">
   398→        <span className="text-xs text-ub-text-muted">Time to Close</span>
   399→        <span className={`text-sm font-mono font-semibold ${isUrgent ? 'text-ub-warning' : 'text-ub-text-primary'}`}>
   400→          {timeToClose}
   401→        </span>
   402→      </div>
   403→    </div>
   404→  );
