   177→// ─────────────────────────────────────────────
   178→// Indian Number Formatting
   179→// ─────────────────────────────────────────────
   180→
   181→function formatINR(value: number): string {
   182→  const sign = value < 0 ? '-' : '';
   183→  const abs = Math.abs(value);
   184→  const parts = abs.toFixed(2).split('.');
   185→  const intPart = parts[0];
   186→  const decPart = parts[1];
   187→  let formatted = '';
   188→  if (intPart.length <= 3) {
   189→    formatted = intPart;
   190→  } else {
   191→    const last3 = intPart.slice(-3);
   192→    const rest = intPart.slice(0, -3);
   193→    formatted = rest.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',' + last3;
   194→  }
   195→  return `₹${sign}${formatted}.${decPart}`;
   196→}
   197→
   198→function formatPercent(value: number): string {
   199→  const sign = value >= 0 ? '+' : '';
   200→  return `${sign}${value.toFixed(2)}%`;
   201→}

   203→// ─────────────────────────────────────────────
   204→// Circular Progress (SVG)
   205→// ─────────────────────────────────────────────
   206→
   207→function CircularProgress({ value, size = 64, strokeWidth = 5, color = '#00d09c' }: {
   208→  value: number;
   209→  size?: number;
   210→  strokeWidth?: number;
   211→  color?: string;
   212→}) {
   213→  const radius = (size - strokeWidth) / 2;
   214→  const circumference = 2 * Math.PI * radius;
   215→  const offset = circumference - (value / 100) * circumference;
   216→
   217→  return (
   218→    <svg width={size} height={size} className="-rotate-90">
   219→      <circle
   220→        cx={size / 2}
   221→        cy={size / 2}
   222→        r={radius}
   223→        fill="none"
   224→        stroke="#1e293b"
   225→        strokeWidth={strokeWidth}
   226→      />
   227→      <circle
   228→        cx={size / 2}
   229→        cy={size / 2}
   230→        r={radius}
   231→        fill="none"
   232→        stroke={color}
   233→        strokeWidth={strokeWidth}
   234→        strokeDasharray={circumference}
   235→        strokeDashoffset={offset}
   236→        strokeLinecap="round"
   237→        className="transition-all duration-700 ease-out"
   238→      />
   239→      <text
   240→        x={size / 2}
   241→        y={size / 2}
   242→        className="fill-ub-text-primary text-xs font-bold"
   243→        textAnchor="middle"
   244→        dominantBaseline="central"
   245→        transform={`rotate(90, ${size / 2}, ${size / 2})`}
   246→      >
   247→        {value.toFixed(0)}%
   248→      </text>
   249→    </svg>
   250→  );
   251→}

   355→// ─────────────────────────────────────────────
   356→// Market Timer Component
   357→// ─────────────────────────────────────────────
   358→
   359→function MarketTimer() {
   360→  const [now, setNow] = useState(() => new Date());
   361→  const engine = useEngineStore();
   362→
   363→  useEffect(() => {
   364→    const interval = setInterval(() => setNow(new Date()), 1_000);
   365→    return () => clearInterval(interval);
   366→  }, []);
   367→
   368→  const istString = now.toLocaleTimeString('en-IN', {
   369→    timeZone: 'Asia/Kolkata',
   370→    hour12: false,
   371→    hour: '2-digit',
   372→    minute: '2-digit',
   373→    second: '2-digit',
   374→  });
   375→
   376→  const marketCloseSeconds = engine.marketCloseSeconds;
   377→  const hours = Math.floor(marketCloseSeconds / 3600);
   378→  const minutes = Math.floor((marketCloseSeconds % 3600) / 60);
   379→  const seconds = Math.floor(marketCloseSeconds % 60);
   380→  const pad = (n: number) => String(n).padStart(2, '0');
   381→  const timeToClose = marketCloseSeconds > 0
   382→    ? `${pad(hours)}:${pad(minutes)}:${pad(seconds)}`
   383→    : 'Market Closed';
   384→
   385→  const isUrgent = marketCloseSeconds > 0 && marketCloseSeconds < 1800;
   386→
   387→  return (
   388→    <div className="flex flex-col gap-2">
   389→      <div className="flex items-center gap-2">
   390→        <Clock className="h-4 w-4 text-ub-text-muted" />
   391→        <span className="text-lg font-mono font-bold text-ub-text-primary tracking-wider">
   392→          {istString}
   393→        </span>
   394→        <Badge variant="outline" className="text-[10px] font-medium border-ub-border text-ub-text-muted">
   395→          IST
   396→        </Badge>
   397→      </div>
   398→      <Separator className="bg-ub-border" />
   399→      <div className="flex items-center justify-between">
   400→        <span className="text-xs text-ub-text-muted">Time to Close</span>
   401→        <span className={`text-sm font-mono font-semibold ${isUrgent ? 'text-ub-warning' : 'text-ub-text-primary'}`}>
   402→          {timeToClose}
   403→        </span>
   404→      </div>
   405→    </div>
   406→  );
   407→}
