    70→   207→    async def get_todays_pnl(self) -> Dict[str, Any]:
    71→   208→        """Get today's aggregate P&L."""
    72→   209→        today = _today_str()
    73→   210→        trades = await self.get_trades_by_date(today, limit=500)
    74→   211→        closed = [t for t in trades if t.status == "CLOSED"]
    75→   212→        gross_pnl = sum(t.pnl for t in closed)
    76→   213→        total_fees = sum(t.fees for t in closed) + sum(t.brokerage for t in closed)
    77→   214→        net_pnl = sum(t.net_pnl for t in closed)
    78→   215→        wins = sum(1 for t in closed if t.net_pnl > 0)
    79→   216→        losses = sum(1 for t in closed if t.net_pnl < 0)
    80→   217→        total = len(closed)
    81→   218→        return {
    82→   219→            "date": today,
    83→   220→            "total_trades": total,
    84→   221→            "wins": wins,
    85→   222→            "losses": losses,
    86→   223→            "win_rate": (wins / total * 100) if total > 0 else 0.0,
    87→   224→            "gross_pnl": round(gross_pnl, 2),
    88→   225→            "total_fees": round(total_fees, 2),
    89→   226→            "net_pnl": round(net_pnl, 2),
    90→   227→            "best_trade": round(max((t.net_pnl for t in closed), default=0), 2),
    91→   228→            "worst_trade": round(min((t.net_pnl for t in closed), default=0), 2),
    92→   229→        }

   624→   761→    async def get_max_drawdown_pct(self) -> float:
   625→   762→        """Calculate max drawdown percentage from all closed trades."""
   626→   763→        stmt = select(Trade).where(Trade.status == "CLOSED").order_by(Trade.exit_time.asc())
   627→   764→        result = await self.session.execute(stmt)
   628→   765→        trades = list(result.scalars().all())
   629→   766→        if not trades:
   630→   767→            return 0.0
   631→   768→        peak = 100000.0  # Starting capital
   632→   769→        max_dd = 0.0
   633→   770→        running = peak
   634→   771→        for t in trades:
   635→   772→            running += t.net_pnl
   636→   773→            if running > peak:
   637→   774→                peak = running
   638→   775→            dd = (peak - running) / peak * 100 if peak > 0 else 0.0
   639→   776→            if dd > max_dd:
   640→   777→                max_dd = dd
   641→   778→        return round(max_dd, 2)
