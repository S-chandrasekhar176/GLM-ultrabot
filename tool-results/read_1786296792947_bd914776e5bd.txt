   108→   207→    async def get_todays_pnl(self) -> Dict[str, Any]:
   109→   208→        """Get today's aggregate P&L."""
   110→   209→        today = _today_str()
   111→   210→        trades = await self.get_trades_by_date(today, limit=500)
   112→   211→        closed = [t for t in trades if t.status == "CLOSED"]
   113→   212→        gross_pnl = sum(t.pnl for t in closed)
   114→   213→        total_fees = sum(t.fees for t in closed) + sum(t.brokerage for t in closed)
   115→   214→        net_pnl = sum(t.net_pnl for t in closed)
   116→   215→        wins = sum(1 for t in closed if t.net_pnl > 0)
   117→   216→        losses = sum(1 for t in closed if t.net_pnl < 0)
   118→   217→        total = len(closed)
   119→   218→        return {
   120→   219→            "date": today,
   121→   220→            "total_trades": total,
   122→   221→            "wins": wins,
   123→   222→            "losses": losses,
   124→   223→            "win_rate": (wins / total * 100) if total > 0 else 0.0,
   125→   224→            "gross_pnl": round(gross_pnl, 2),
   126→   225→            "total_fees": round(total_fees, 2),
   127→   226→            "net_pnl": round(net_pnl, 2),
   128→   227→            "best_trade": round(max((t.net_pnl for t in closed), default=0), 2),
   129→   228→            "worst_trade": round(min((t.net_pnl for t in closed), default=0), 2),
   130→   229→        }

   662→   761→    async def get_max_drawdown_pct(self) -> float:
   663→   762→        """Calculate max drawdown percentage from all closed trades."""
   664→   763→        stmt = select(Trade).where(Trade.status == "CLOSED").order_by(Trade.exit_time.asc())
   665→   764→        result = await self.session.execute(stmt)
   666→   765→        trades = list(result.scalars().all())
   667→   766→        if not trades:
   668→   767→            return 0.0
   669→   768→        peak = 100000.0  # Starting capital
   670→   769→        max_dd = 0.0
   671→   770→        running = peak
   672→   771→        for t in trades:
   673→   772→            running += t.net_pnl
   674→   773→            if running > peak:
   675→   774→                peak = running
   676→   775→            dd = (peak - running) / peak * 100 if peak > 0 else 0.0
   677→   776→            if dd > max_dd:
   678→   777→                max_dd = dd
   679→   778→        return round(max_dd, 2)
