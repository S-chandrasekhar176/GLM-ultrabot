   237→function getScoreColor(score: number): string {
   238→  if (score >= 0.8) return 'text-ub-profit';

   256→// TimerCountdown Component

   258→
   259→function TimerCountdown({ expiryAt }: { expiryAt: string }) {
   260→  const [timeLeft, setTimeLeft] = useState('');
   261→  const [isExpired, setIsExpired] = useState(false);
   262→
   263→  useEffect(() => {
   264→    const update = () => {
   265→      const diff = new Date(expiryAt).getTime() - Date.now();
   266→      if (diff <= 0) {
   267→        setIsExpired(true);
   268→        setTimeLeft('Expired');
   269→        return;
   270→      }
   271→      const mins = Math.floor(diff / 60_000);
   272→      const secs = Math.floor((diff % 60_000) / 1000);
   273→      setTimeLeft(`${mins}:${secs.toString().padStart(2, '0')}`);
   274→    };
   275→    update();
   276→    const interval = setInterval(update, 1000);
   277→    return () => clearInterval(interval);
   278→  }, [expiryAt]);
   279→
   280→  return (
   281→    <span
   282→      className={`flex items-center gap-1 text-xs font-mono font-medium ${
   283→        isExpired ? 'text-ub-loss' : timeLeft.includes('0:') || timeLeft.includes('1:') ? 'text-ub-warning' : 'text-ub-text-muted'
   284→      }`}
   285→    >
   286→      <Timer className="h-3 w-3" />
   287→      {timeLeft}
   288→    </span>
   289→  );
   290→}

   762→type FilterTab = 'all' | 'pending' | 'confirmed' | 'skipped' | 'expired';
   763→
   764→export default function OpportunitiesPage() {
   765→  const [activeTab, setActiveTab] = useState<FilterTab>('all');
   766→  const [opportunities, setOpportunities] = useState<OpportunityData[]>(MOCK_OPPORTUNITIES);
   767→  const [isLoading] = useState(false);
   768→  const [isConfirming, setIsConfirming] = useState(false);
   769→  const [isSkipping, setIsSkipping] = useState(false);
   770→
   771→  const filtered = useMemo(() => {
   772→    if (activeTab === 'all') return opportunities;
   773→    return opportunities.filter((o) => o.status === activeTab);
   774→  }, [opportunities, activeTab]);
   775→
   776→  const pendingCount = opportunities.filter((o) => o.status === 'pending').length;
   777→
   778→  const handleConfirm = useCallback((id: string) => {
   779→    setIsConfirming(true);
   780→    const opp = opportunities.find((o) => o.id === id);
   781→    // Simulate API call
   782→    setTimeout(() => {
   783→      setOpportunities((prev) =>
   784→        prev.map((o) => (o.id === id ? { ...o, status: 'confirmed' as OppStatus } : o))
   785→      );
   786→      setIsConfirming(false);
   787→      toast.success(`${opp?.symbol} confirmed`, {
   788→        description: `${opp?.strategy} trade has been placed successfully.`,
   789→      });
   790→    }, 800);
   791→  }, [opportunities]);
   792→
   793→  const handleSkip = useCallback((id: string) => {
   794→    setIsSkipping(true);
   795→    const opp = opportunities.find((o) => o.id === id);
   796→    setTimeout(() => {
   797→      setOpportunities((prev) =>
   798→        prev.map((o) => (o.id === id ? { ...o, status: 'skipped' as OppStatus } : o))
   799→      );
   800→      setIsSkipping(false);
   801→      toast.info(`${opp?.symbol} skipped`, {
   802→        description: 'Opportunity has been moved to skipped list.',
   803→      });
   804→    }, 500);
   805→  }, [opportunities]);
   806→
   807→  return (
   808→    <div className="space-y-6">
   809→      {/* Header */}
   810→      <div className="flex flex-col sm:flex-row sm:items-center gap-3">
   811→        <div className="flex items-center gap-3">
   812→          <Target className="h-6 w-6 text-ub-accent" />
   813→          <h1 className="text-2xl font-bold text-ub-text-primary">Opportunities</h1>
   814→          {pendingCount > 0 && (
   815→            <Badge className="bg-ub-accent/15 text-ub-accent border-ub-accent/30 font-semibold">
   816→              {pendingCount}
   817→            </Badge>
   818→          )}
   819→        </div>
   820→        <Tabs
   821→          value={activeTab}
   822→          onValueChange={(v) => setActiveTab(v as FilterTab)}
   823→          className="ml-auto"
   824→        >
   825→          <TabsList className="bg-ub-background border border-ub-border">
   826→            <TabsTrigger
   827→              value="all"
   828→              className="data-[state=active]:bg-ub-surface-active data-[state=active]:text-ub-text-primary text-ub-text-muted text-xs"
   829→            >
   830→              All ({opportunities.length})
   831→            </TabsTrigger>
   832→            <TabsTrigger
   833→              value="pending"
   834→              className="data-[state=active]:bg-ub-surface-active data-[state=active]:text-ub-text-primary text-ub-text-muted text-xs"
   835→            >
   836→              Pending ({pendingCount})
   837→            </TabsTrigger>
   838→            <TabsTrigger
   839→              value="confirmed"
   840→              className="data-[state=active]:bg-ub-surface-active data-[state=active]:text-ub-text-primary text-ub-text-muted text-xs"

   842→              Confirmed ({opportunities.filter((o) => o.status === 'confirmed').length})

   845→              value="skipped"
   846→              className="data-[state=active]:bg-ub-surface-active data-[state=active]:text-ub-text-primary text-ub-text-muted text-xs"
   847→            >
   848→              Skipped ({opportunities.filter((o) => o.status === 'skipped').length})
   849→            </TabsTrigger>
   850→            <TabsTrigger
   851→              value="expired"
   852→              className="data-[state=active]:bg-ub-surface-active data-[state=active]:text-ub-text-primary text-ub-text-muted text-xs"
   853→            >
   854→              Expired ({opportunities.filter((o) => o.status === 'expired').length})
   855→            </TabsTrigger>
   856→          </TabsList>
   857→        </Tabs>
   858→      </div>
   859→
   860→      {/* Content */}
   861→      {isLoading ? (
   862→        <div className="space-y-4">

   864→            <OpportunityCardSkeleton key={i} />
