   176→   176→// Indian Number Formatting
   177→   177→// ─────────────────────────────────────────────
   178→   178→
   179→   179→function formatINR(value: number): string {
   180→   180→  const sign = value < 0 ? '-' : '';
   181→   181→  const abs = Math.abs(value);
   182→   182→  const parts = abs.toFixed(2).split('.');
   183→   183→  const intPart = parts[0];
   184→   184→  const decPart = parts[1];
   185→   185→  let formatted = '';
   186→   186→  if (intPart.length <= 3) {
   187→   187→    formatted = intPart;
   188→   188→  } else {
   189→   189→    const last3 = intPart.slice(-3);
   190→   190→    const rest = intPart.slice(0, -3);
   191→   191→    formatted = rest.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',' + last3;
   192→   192→  }
   193→   193→  return `₹${sign}${formatted}.${decPart}`;
   194→   194→}
   195→   195→
   196→   196→function formatPercent(value: number): string {
   197→   197→  const sign = value >= 0 ? '+' : '';
   198→   198→  return `${sign}${value.toFixed(2)}%`;
   199→   199→}

   205→   205→function CircularProgress({ value, size = 64, strokeWidth = 5, color = '#00d09c' }: {
   206→   206→  value: number;
   207→   207→  size?: number;
   208→   208→  strokeWidth?: number;

   210→   210→}) {

   212→   212→  const circumference = 2 * Math.PI * radius;
   213→   213→  const offset = circumference - (value / 100) * circumference;
   214→   214→
   215→   215→  return (
   216→   216→    <svg width={size} height={size} className="-rotate-90">
   217→   217→      <circle
   218→   218→        cx={size / 2}
   219→   219→        cy={size / 2}
   220→   220→        r={radius}
   221→   221→        fill="none"
   222→   222→        stroke="#1e293b"

   224→   224→      />

   357→   357→function MarketTimer() {
   358→   358→  const [now, setNow] = useState(() => new Date());
   359→   359→  const engine = useEngineStore();
   360→   360→
   361→   361→  useEffect(() => {
   362→   362→    const interval = setInterval(() => setNow(new Date()), 1_000);
   363→   363→    return () => clearInterval(interval);
   364→   364→  }, []);
   365→   365→
   366→   366→  const istString = now.toLocaleTimeString('en-IN', {
   367→   367→    timeZone: 'Asia/Kolkata',
   368→   368→    hour12: false,
   369→   369→    hour: '2-digit',
   370→   370→    minute: '2-digit',
   371→   371→    second: '2-digit',
   372→   372→  });
   373→   373→
   374→   374→  const marketCloseSeconds = engine.marketCloseSeconds;
   375→   375→  const hours = Math.floor(marketCloseSeconds / 3600);
   376→   376→  const minutes = Math.floor((marketCloseSeconds % 3600) / 60);
   377→   377→  const seconds = Math.floor(marketCloseSeconds % 60);
   378→   378→  const pad = (n: number) => String(n).padStart(2, '0');
   379→   379→  const timeToClose = marketCloseSeconds > 0
   380→   380→    ? `${pad(hours)}:${pad(minutes)}:${pad(seconds)}`
   381→   381→    : 'Market Closed';
   382→   382→
   383→   383→  const isUrgent = marketCloseSeconds > 0 && marketCloseSeconds < 1800;
   384→   384→
   385→   385→  return (
   386→   386→    <div className="flex flex-col gap-2">
   387→   387→      <div className="flex items-center gap-2">
   388→   388→        <Clock className="h-4 w-4 text-ub-text-muted" />
   389→   389→        <span className="text-lg font-mono font-bold text-ub-text-primary tracking-wider">
   390→   390→          {istString}
   391→   391→        </span>
   392→   392→        <Badge variant="outline" className="text-[10px] font-medium border-ub-border text-ub-text-muted">
   393→   393→          IST
