   173→   173→function generateMockResult(form: BacktestForm): BacktestResult {
   174→   174→  const totalTrades = Math.floor(Math.random() * 80) + 40;
   175→   175→  const winRate = 55 + Math.random() * 20;
   176→   176→  const winningTrades = Math.floor(totalTrades * winRate / 100);
   177→   177→  const losingTrades = totalTrades - winningTrades;
   178→   178→  const avgWin = (Math.random() * 3000 + 1500);
   179→   179→  const avgLoss = (Math.random() * 1500 + 500);
   180→   180→  const totalProfit = winningTrades * avgWin;
   181→   181→  const totalLoss = losingTrades * avgLoss;
   182→   182→  const totalPnl = totalProfit - totalLoss;
   183→   183→  const totalReturn = (totalPnl / form.capital) * 100;
   184→   184→  const annualizedReturn = totalReturn * (365 / 365);
   185→   185→  const maxDrawdown = -(Math.random() * 15 + 5);
   186→   186→  const sharpe = (Math.random() * 2 + 0.5);
   187→   187→  const sortino = sharpe * (1 + Math.random() * 0.5);
   188→   188→  const profitFactor = totalProfit / totalLoss;
   189→   189→  const totalFees = form.includeFees ? totalTrades * (Math.random() * 50 + 20) : 0;
   190→   190→  const avgTradePnl = totalPnl / totalTrades;
   191→   191→
   192→   192→  // Equity curve
   193→   193→  const equityCurve: EquityPoint[] = [];
   194→   194→  let equity = form.capital;
   195→   195→  for (let i = 0; i <= totalTrades; i++) {
   196→   196→    if (i > 0) {
   197→   197→      const isWin = Math.random() * 100 < winRate;
   198→   198→      equity += isWin ? avgWin * (0.5 + Math.random()) : -(avgLoss * (0.5 + Math.random()));
   199→   199→    }
   200→   200→    equityCurve.push({ trade: i, equity: Math.round(equity) });
   201→   201→  }
   202→   202→
   203→   203→  // Drawdown chart
   204→   204→  const drawdownChart: DrawdownPoint[] = [];
   205→   205→  let peak = form.capital;
   206→   206→  const dates: string[] = [];
   207→   207→  for (let i = 0; i < totalTrades; i++) {
   208→   208→    const d = new Date(2025, 0, 1 + i * 3);
   209→   209→    dates.push(d.toISOString().split('T')[0]);
   210→   210→  }
   211→   211→  for (let i = 0; i < totalTrades; i++) {
   212→   212→    const eq = equityCurve[i + 1]?.equity ?? equity;
   213→   213→    peak = Math.max(peak, eq);
   214→   214→    const dd = ((eq - peak) / peak) * 100;
   215→   215→    drawdownChart.push({ date: dates[i] || `Day ${i}`, drawdown: Math.round(dd * 100) / 100 });
   216→   216→  }
   217→   217→
   218→   218→  // Monthly returns
   219→   219→  const monthlyReturns: MonthlyReturn[] = [];
   220→   220→  for (let m = 0; m < 12; m++) {
   221→   221→    monthlyReturns.push({
   222→   222→      month: MONTHS[m],
   223→   223→      year: 2025,
   224→   224→      value: Math.round((Math.random() * 16 - 6) * 100) / 100,
   225→   225→    });
   226→   226→  }
   227→   227→
   228→   228→  // Trades
   229→   229→  const symbols = form.symbols.split(',').map(s => s.trim().toUpperCase()).filter(Boolean);
   230→   230→  const trades: Trade[] = [];
   231→   231→  const exitReasons = ['Target Hit', 'Stop Loss', 'Signal Reverse', 'EOD Square-off', 'Trailing SL'];
   232→   232→  for (let i = 0; i < Math.min(totalTrades, 50); i++) {
   233→   233→    const isBuy = Math.random() > 0.5;
   234→   234→    const entry = Math.random() * 3000 + 200;
   235→   235→    const isWin = Math.random() * 100 < winRate;
   236→   236→    const pnlVal = isWin
   237→   237→      ? (Math.random() * 4000 + 500)
   238→   238→      : -(Math.random() * 2000 + 200);
   239→   239→    const exit = isBuy ? entry + pnlVal / 100 : entry - pnlVal / 100;
   240→   240→    const durH = Math.floor(Math.random() * 8 + 0.5);
   241→   241→    const durM = Math.floor(Math.random() * 59);
   242→   242→    const tradeDate = new Date(2025, 0, 2 + i * 2);
   243→   243→
   244→   244→    trades.push({
   245→   245→      id: i + 1,
   246→   246→      date: tradeDate.toLocaleDateString('en-IN'),
   247→   247→      symbol: symbols[Math.floor(Math.random() * symbols.length)] || 'RELIANCE',
   248→   248→      direction: isBuy ? 'BUY' : 'SELL',
   249→   249→      entry: Math.round(entry * 100) / 100,
   250→   250→      exit: Math.round(exit * 100) / 100,
   251→   251→      pnl: Math.round(pnlVal),
   252→   252→      duration: `${durH}h ${durM}m`,
   253→   253→      exitReason: exitReasons[Math.floor(Math.random() * exitReasons.length)],
   254→   254→    });
   255→   255→  }
   256→   256→
   257→   257→  // Risk gate stats
   258→   258→  const riskGateStats: RiskGateStat[] = [
   259→   259→    { name: 'VIX Filter', passed: Math.floor(Math.random() * 30 + 20), rejected: Math.floor(Math.random() * 10 + 2) },
   260→   260→    { name: 'Daily Loss Limit', passed: Math.floor(Math.random() * 25 + 25), rejected: Math.floor(Math.random() * 8 + 1) },
   261→   261→    { name: 'Position Limit', passed: Math.floor(Math.random() * 28 + 15), rejected: Math.floor(Math.random() * 6 + 3) },
   262→   262→    { name: 'Confidence Filter', passed: Math.floor(Math.random() * 22 + 20), rejected: Math.floor(Math.random() * 12 + 2) },
   263→   263→    { name: 'Sector Exposure', passed: Math.floor(Math.random() * 30 + 18), rejected: Math.floor(Math.random() * 5 + 1) },
   264→   264→  ];
   265→   265→
   266→   266→  return {
   267→   267→    metrics: {
   268→   268→      totalReturn: Math.round(totalReturn * 100) / 100,
   269→   269→      annualizedReturn: Math.round(annualizedReturn * 100) / 100,
   270→   270→      maxDrawdown: Math.round(maxDrawdown * 100) / 100,
   271→   271→      sharpe: Math.round(sharpe * 100) / 100,
   272→   272→      sortino: Math.round(sortino * 100) / 100,
   273→   273→      winRate: Math.round(winRate * 100) / 100,
   274→   274→      profitFactor: Math.round(profitFactor * 100) / 100,
   275→   275→      totalTrades,
   276→   276→      totalFees: Math.round(totalFees),
   277→   277→      avgTradePnl: Math.round(avgTradePnl),
   278→   278→    },
   279→   279→    equityCurve,
   280→   280→    drawdownChart,
   281→   281→    monthlyReturns,
   282→   282→    trades,
   283→   283→    riskGateStats,
   284→   284→  };
   285→   285→}
