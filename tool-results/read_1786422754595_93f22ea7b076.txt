   142→   181→   181→   181→function formatINR(value: number): string {
   143→   182→   182→   182→  const sign = value < 0 ? '-' : '';
   144→   183→   183→   183→  const abs = Math.abs(value);
   145→   184→   184→   184→  const parts = abs.toFixed(2).split('.');
   146→   185→   185→   185→  const intPart = parts[0];
   147→   186→   186→   186→  const decPart = parts[1];
   148→   187→   187→   187→  let formatted = '';
   149→   188→   188→   188→  if (intPart.length <= 3) {
   150→   189→   189→   189→    formatted = intPart;
   151→   190→   190→   190→  } else {
   152→   191→   191→   191→    const last3 = intPart.slice(-3);
   153→   192→   192→   192→    const rest = intPart.slice(0, -3);
   154→   193→   193→   193→    formatted = rest.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',' + last3;
   155→   194→   194→   194→  }
   156→   195→   195→   195→  return `₹${sign}${formatted}.${decPart}`;
   157→   196→   196→   196→}
