   483→// ─────────────────────────────────────────────
   484→// Positions Tab Component
   485→// ─────────────────────────────────────────────
   486→
   487→function PositionsTab() {
   488→  const [positions, setPositions] = useState<Position[]>(MOCK_POSITIONS);
   489→  const [closeDialogId, setCloseDialogId] = useState<string | null>(null);
   490→  const [modifyDialog, setModifyDialog] = useState<{
   491→    open: boolean;
   492→    id: string;
   493→    field: 'stopLoss' | 'target';
   494→  }>({ open: false, id: '', field: 'stopLoss' });
   495→  const [modifyValue, setModifyValue] = useState('');
   496→  const [partialDialog, setPartialDialog] = useState<{
   497→    open: boolean;
   498→    id: string;
   499→  }>({ open: false, id: '' });
   500→  const [partialQty, setPartialQty] = useState('');
   501→
   502→  const totalUnrealizedPnl = useMemo(
   503→    () => positions.reduce((sum, p) => sum + p.unrealizedPnl, 0),
   504→    [positions]
   505→  );
   506→
   507→  const totalInvested = useMemo(
   508→    () => positions.reduce((sum, p) => sum + p.entry * p.quantity, 0),
   509→    [positions]
   510→  );
   511→
   512→  const handleClosePosition = (id: string) => {
   513→    setPositions((prev) => prev.filter((p) => p.id !== id));
   514→    setCloseDialogId(null);
   515→    const pos = positions.find((p) => p.id === id);
   516→    toast.success(`${pos?.symbol} position closed`, {
   517→      description: `P&L: ${INR(pos?.unrealizedPnl ?? 0)}`,
   518→    });
   519→  };
   520→
   521→  const handleModify = () => {
   522→    const val = parseFloat(modifyValue);
   523→    if (isNaN(val) || val <= 0) return;
   524→    setPositions((prev) =>
   525→      prev.map((p) =>
   526→        p.id === modifyDialog.id
   527→          ? { ...p, [modifyDialog.field]: val }
   528→          : p
   529→      )
   530→    );
   531→    const pos = positions.find((p) => p.id === modifyDialog.id);
   532→    toast.success(`${pos?.symbol} ${modifyDialog.field === 'stopLoss' ? 'SL' : 'Target'} updated`, {
   533→      description: `New ${modifyDialog.field === 'stopLoss' ? 'Stop Loss' : 'Target'}: ${INR(val)}`,
   534→    });
   535→    setModifyDialog({ open: false, id: '', field: 'stopLoss' });
   536→    setModifyValue('');
   537→  };
   538→
   539→  const handlePartialClose = () => {
   540→    const qty = parseInt(partialQty, 10);
   541→    if (isNaN(qty) || qty <= 0) return;
   542→    setPositions((prev) =>
   543→      prev.map((p) =>
   544→        p.id === partialDialog.id
   545→          ? {
   546→              ...p,
   547→              remainingQty: Math.max(0, p.remainingQty - qty),
   548→              quantity: p.quantity,
   549→            }
   550→          : p
   551→      )
   552→    );
   553→    const pos = positions.find((p) => p.id === partialDialog.id);
   554→    toast.success(`Partial close on ${pos?.symbol}`, {
   555→      description: `${qty} shares closed, ${Math.max(0, (pos?.remainingQty ?? 0) - qty)} remaining.`,
   556→    });
   557→    setPartialDialog({ open: false, id: '' });
   558→    setPartialQty('');
   559→  };
   560→
   561→  const openModify = (id: string, field: 'stopLoss' | 'target') => {
   562→    const pos = positions.find((p) => p.id === id);
   563→ setModifyValue(pos ? String(pos[field]) : '');
   564→    setModifyDialog({ open: true, id, field });
   565→  };
   566→
   567→  const openPartial = (id: string) => {
   568→    const pos = positions.find((p) => p.id === id);
   569→    setPartialQty(pos ? String(pos.remainingQty) : '');
   570→    setPartialDialog({ open: true, id });
   571→  };
   572→
   573→  return (
   574→    <div className="space-y-4">
   575→      {/* Summary Bar */}
   576→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   577→        <Card className="bg-ub-surface border-ub-border">
   578→          <CardContent className="p-4 flex items-center gap-3">
   579→            <div className="h-9 w-9 rounded-lg bg-ub-accent/10 flex items-center justify-center">
   580→              <ArrowLeftRight className="h-4.5 w-4.5 text-ub-accent" />
   581→            </div>
   582→            <div>
   583→              <p className="text-[11px] uppercase tracking-wider text-ub-text-muted">Total Positions</p>
   584→              <p className="text-xl font-bold text-ub-text-primary">{positions.length}</p>
   585→            </div>
   586→          </CardContent>
   587→        </Card>
   588→        <Card className="bg-ub-surface border-ub-border">
   589→          <CardContent className="p-4 flex items-center gap-3">
   590→            <div
   591→              className={`h-9 w-9 rounded-lg flex items-center justify-center ${
   592→                totalUnrealizedPnl >= 0 ? 'bg-ub-profit/10' : 'bg-ub-loss/10'
   593→              }`}
   594→            >
   595→              {totalUnrealizedPnl >= 0 ? (
   596→                <TrendingUp className="h-4.5 w-4.5 text-ub-profit" />
   597→              ) : (
   598→                <TrendingDown className="h-4.5 w-4.5 text-ub-loss" />
   599→              )}
   600→            </div>
   601→            <div>
   602→              <p className="text-[11px] uppercase tracking-wider text-ub-text-muted">Unrealized P&L</p>
   603→              <p
   604→                className={`text-xl font-bold ${
   605→                  totalUnrealizedPnl >= 0 ? 'text-ub-profit' : 'text-ub-loss'
   606→                }`}
   607→              >
   608→                {INR(totalUnrealizedPnl)}
   609→              </p>
   610→            </div>
   611→          </CardContent>
   612→        </Card>
   613→        <Card className="bg-ub-surface border-ub-border">
   614→          <CardContent className="p-4 flex items-center gap-3">
   615→            <div className="h-9 w-9 rounded-lg bg-ub-warning/10 flex items-center justify-center">
   616→              <Target className="h-4.5 w-4.5 text-ub-warning" />
   617→            </div>
   618→            <div>
   619→              <p className="text-[11px] uppercase tracking-wider text-ub-text-muted">Total Invested</p>
   620→              <p className="text-xl font-bold text-ub-text-primary">{INR_SHORT(totalInvested)}</p>
   621→            </div>
