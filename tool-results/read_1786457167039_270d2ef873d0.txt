   105→   173→function generateMockResult(form: BacktestForm): BacktestResult {
   106→   174→  const totalTrades = Math.floor(Math.random() * 80) + 40;
   107→   175→  const winRate = 55 + Math.random() * 20;
   108→   176→  const winningTrades = Math.floor(totalTrades * winRate / 100);
   109→   177→  const losingTrades = totalTrades - winningTrades;
   110→   178→  const avgWin = (Math.random() * 3000 + 1500);
   111→   179→  const avgLoss = (Math.random() * 1500 + 500);
   112→   180→  const totalProfit = winningTrades * avgWin;
   113→   181→  const totalLoss = losingTrades * avgLoss;
   114→   182→  const totalPnl = totalProfit - totalLoss;
   115→   183→  const totalReturn = (totalPnl / form.capital) * 100;
   116→   184→  const annualizedReturn = totalReturn * (365 / 365);
   117→   185→  const maxDrawdown = -(Math.random() * 15 + 5);
   118→   186→  const sharpe = (Math.random() * 2 + 0.5);
   119→   187→  const sortino = sharpe * (1 + Math.random() * 0.5);
   120→   188→  const profitFactor = totalProfit / totalLoss;
   121→   189→  const totalFees = form.includeFees ? totalTrades * (Math.random() * 50 + 20) : 0;
   122→   190→  const avgTradePnl = totalPnl / totalTrades;
   123→   191→
   124→   192→  // Equity curve
   125→   193→  const equityCurve: EquityPoint[] = [];
   126→   194→  let equity = form.capital;
   127→   195→  for (let i = 0; i <= totalTrades; i++) {
   128→   196→    if (i > 0) {
   129→   197→      const isWin = Math.random() * 100 < winRate;
   130→   198→      equity += isWin ? avgWin * (0.5 + Math.random()) : -(avgLoss * (0.5 + Math.random()));
   131→   199→    }
   132→   200→    equityCurve.push({ trade: i, equity: Math.round(equity) });
   133→   201→  }
   134→   202→
   135→   203→  // Drawdown chart
   136→   204→  const drawdownChart: DrawdownPoint[] = [];
   137→   205→  let peak = form.capital;
   138→   206→  const dates: string[] = [];
   139→   207→  for (let i = 0; i < totalTrades; i++) {
   140→   208→    const d = new Date(2025, 0, 1 + i * 3);
   141→   209→    dates.push(d.toISOString().split('T')[0]);
   142→   210→  }
   143→   211→  for (let i = 0; i < totalTrades; i++) {
   144→   212→    const eq = equityCurve[i + 1]?.equity ?? equity;
   145→   213→    peak = Math.max(peak, eq);
   146→   214→    const dd = ((eq - peak) / peak) * 100;
   147→   215→    drawdownChart.push({ date: dates[i] || `Day ${i}`, drawdown: Math.round(dd * 100) / 100 });
   148→   216→  }
   149→   217→
   150→   218→  // Monthly returns
   151→   219→  const monthlyReturns: MonthlyReturn[] = [];
   152→   220→  for (let m = 0; m < 12; m++) {
   153→   221→    monthlyReturns.push({
   154→   222→      month: MONTHS[m],
   155→   223→      year: 2025,
   156→   224→      value: Math.round((Math.random() * 16 - 6) * 100) / 100,
   157→   225→    });
   158→   226→  }
   159→   227→
   160→   228→  // Trades
   161→   229→  const symbols = form.symbols.split(',').map(s => s.trim().toUpperCase()).filter(Boolean);
   162→   230→  const trades: Trade[] = [];
   163→   231→  const exitReasons = ['Target Hit', 'Stop Loss', 'Signal Reverse', 'EOD Square-off', 'Trailing SL'];
   164→   232→  for (let i = 0; i < Math.min(totalTrades, 50); i++) {
   165→   233→    const isBuy = Math.random() > 0.5;
   166→   234→    const entry = Math.random() * 3000 + 200;
   167→   235→    const isWin = Math.random() * 100 < winRate;
   168→   236→    const pnlVal = isWin
   169→   237→      ? (Math.random() * 4000 + 500)
   170→   238→      : -(Math.random() * 2000 + 200);
   171→   239→    const exit = isBuy ? entry + pnlVal / 100 : entry - pnlVal / 100;
   172→   240→    const durH = Math.floor(Math.random() * 8 + 0.5);
   173→   241→    const durM = Math.floor(Math.random() * 59);
   174→   242→    const tradeDate = new Date(2025, 0, 2 + i * 2);
   175→   243→
   176→   244→    trades.push({
   177→   245→      id: i + 1,
   178→   246→      date: tradeDate.toLocaleDateString('en-IN'),
   179→   247→      symbol: symbols[Math.floor(Math.random() * symbols.length)] || 'RELIANCE',
   180→   248→      direction: isBuy ? 'BUY' : 'SELL',
   181→   249→      entry: Math.round(entry * 100) / 100,
   182→   250→      exit: Math.round(exit * 100) / 100,
   183→   251→      pnl: Math.round(pnlVal),
   184→   252→      duration: `${durH}h ${durM}m`,
   185→   253→      exitReason: exitReasons[Math.floor(Math.random() * exitReasons.length)],
   186→   254→    });
   187→   255→  }
   188→   256→
   189→   257→  // Risk gate stats
   190→   258→  const riskGateStats: RiskGateStat[] = [
   191→   259→    { name: 'VIX Filter', passed: Math.floor(Math.random() * 30 + 20), rejected: Math.floor(Math.random() * 10 + 2) },
   192→   260→    { name: 'Daily Loss Limit', passed: Math.floor(Math.random() * 25 + 25), rejected: Math.floor(Math.random() * 8 + 1) },
   193→   261→    { name: 'Position Limit', passed: Math.floor(Math.random() * 28 + 15), rejected: Math.floor(Math.random() * 6 + 3) },
   194→   262→    { name: 'Confidence Filter', passed: Math.floor(Math.random() * 22 + 20), rejected: Math.floor(Math.random() * 12 + 2) },
   195→   263→    { name: 'Sector Exposure', passed: Math.floor(Math.random() * 30 + 18), rejected: Math.floor(Math.random() * 5 + 1) },
   196→   264→  ];
   197→   265→
   198→   266→  return {
   199→   267→    metrics: {
   200→   268→      totalReturn: Math.round(totalReturn * 100) / 100,
   201→   269→      annualizedReturn: Math.round(annualizedReturn * 100) / 100,
   202→   270→      maxDrawdown: Math.round(maxDrawdown * 100) / 100,
   203→   271→      sharpe: Math.round(sharpe * 100) / 100,
   204→   272→      sortino: Math.round(sortino * 100) / 100,
   205→   273→      winRate: Math.round(winRate * 100) / 100,
   206→   274→      profitFactor: Math.round(profitFactor * 100) / 100,
   207→   275→      totalTrades,
   208→   276→      totalFees: Math.round(totalFees),
   209→   277→      avgTradePnl: Math.round(avgTradePnl),
   210→   278→    },
   211→   279→    equityCurve,
   212→   280→    drawdownChart,
