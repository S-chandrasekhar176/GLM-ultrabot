   411→   910→   910→## 3.5 Dynamic Position Sizing — Detailed Logic
   412→   911→   911→
   413→   912→   912→```
   414→   913→   913→STEP 1: BASE KELLY FRACTION
   415→   914→   914→━━━━━━━━━━━━━━━━━━━━━━━━━━━
   416→   915→   915→  win_rate     = from strategy_performance table (historical)
   417→   916→   916→  avg_win      = average winning trade pnl
   418→   917→   917→  avg_loss     = average losing trade pnl
   419→   918→   918→  kelly_f      = (win_rate × avg_win - (1 - win_rate) × avg_loss) / avg_win
   420→   919→   919→  kelly_f      = clamp(kelly_f, 0.02, 0.25)  // Never below 2%, never above 25%
   421→   920→   920→  
   422→   921→   921→  base_size    = total_capital × kelly_f
   423→   922→   922→
   424→   923→   923→
   425→   924→   924→STEP 2: CONFIDENCE ADJUSTMENT
   426→   925→   925→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   427→   926→   926→  kronos_score = signal.confidence  (0 to 1)
   428→   927→   927→  
   429→   928→   928→  if kronos_score >= 0.8:   confidence_mult = 1.0    (strong signal)
   430→   929→   929→  if kronos_score >= 0.6:   confidence_mult = 0.8    (decent signal)
   431→   930→   930→  if kronos_score >= 0.4:   confidence_mult = 0.5    (weak signal)
   432→   931→   931→  if kronos_score <  0.4:   REJECT by G10           (too weak)
   433→   932→   932→  
   434→   933→   933→  adjusted_size = base_size × confidence_mult
   435→   934→   934→
   436→   935→   935→
   437→   936→   936→STEP 3: VOLATILITY ADJUSTMENT
   438→   937→   937→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   439→   938→   938→  vix = current_vix
   440→   939→   939→  
   441→   940→   940→  if vix <= 14:   vol_mult = 1.0    (calm market, normal size)
   442→   941→   941→  if vix <= 18:   vol_mult = 0.85   (slightly nervous)
   443→   942→   942→  if vix <= 22:   vol_mult = 0.65   (worried, reduce)
   444→   943→   943→  if vix >  22:   vol_mult = 0.4    (fearful, much smaller)
   445→   944→   944→  
   446→   945→   945→  adjusted_size = adjusted_size × vol_mult
   447→   946→   946→
   448→   947→   947→
   449→   948→   948→STEP 4: DRAWDOWN ADJUSTMENT
   450→   949→   949→━━━━━━━━━━━━━━━━━━━━━━━━━━━
   451→   950→   950→  daily_pnl_pct = day_pnl / total_capital
   452→   951→   951→  
   453→   952→   952→  if daily_pnl_pct >= 0:      dd_mult = 1.0    (profitable day, normal)
   454→   953→   953→  if daily_pnl_pct >= -1%:    dd_mult = 0.9    (small loss, slight reduce)
   455→   954→   954→  if daily_pnl_pct >= -2%:    dd_mult = 0.7    (moderate loss, reduce)
   456→   955→   955→  if daily_pnl_pct >= -3%:    dd_mult = 0.4    (approaching limit, heavy reduce)
   457→   956→   956→  if daily_pnl_pct <  -3%:    STOP new trades  (limit hit)
   458→   957→   957→  
   459→   958→   958→  adjusted_size = adjusted_size × dd_mult
   460→   959→   959→
   461→   960→   960→
   462→   961→   961→STEP 5: CAPITAL AVAILABILITY
   463→   962→   962→━━━━━━━━━━━━━━━━━━━━━━━━━━━
   464→   963→   963→  available = free_capital × 0.9   (never use 100%)
   465→   964→   964→  
   466→   965→   965→  if adjusted_size <= available:
   467→   966→   966→      final_size = adjusted_size   (we can afford it)
   468→   967→   967→  else:
   469→   968→   968→      final_size = available       (cap at what we can afford)
   470→   969→   969→  
   471→   970→   970→  if final_size < min_position_size (e.g., ₹5,000):
   472→   971→   971→      REJECT: "Position too small after adjustments"
   473→   972→   972→
   474→   973→   973→
   475→   974→   974→STEP 6: CONVERT TO QUANTITY
   476→   975→   975→━━━━━━━━━━━━━━━━━━━━━━━━━━━
   477→   976→   976→  qty = floor(final_size / entry_price)
   478→   977→   977→  
   479→   978→   978→  if segment == "OPT":
   480→   979→   979→      lot_size = option_lot_size[symbol]  (e.g., RELIANCE lot = 250)
   481→   980→   980→      qty = floor(final_size / (premium × lot_size)) × lot_size
   482→   981→   981→  
   483→   982→   982→  final_capital_required = qty × entry_price  (or qty × premium × lot_size for options)
   484→   983→   983→```

   488→   987→   987→## 3.6 Partial Profit Booking — Detailed Logic
   489→   988→   988→
   490→   989→   989→```
   491→   990→   990→WHEN POSITION IS OPENED:
   492→   991→   991→━━━━━━━━━━━━━━━━━━━━━━━━
   493→   992→   992→  entry  = position.entry_price
   494→   993→   993→  sl     = position.sl_price
   495→   994→   994→  target = position.target_price
   496→   995→   995→  qty    = position.total_quantity
   497→   996→   996→  
   498→   997→   997→  risk   = |entry - sl|                          (risk per share)
   499→   998→   998→  
   500→   999→   999→  level1_target = entry + (risk × 1.0)          (1:1 R:R)
   501→  1000→  1000→  level2_target = entry + (risk × 1.5)          (1.5:1 R:R)
   502→  1001→  1001→  level3_target = entry + (risk × 2.0)          (2:1 R:R) = original target
   503→  1002→  1002→  
   504→  1003→  1003→  (For SELL positions, subtract instead of add)
   505→  1004→  1004→
   506→  1005→  1005→
   507→  1006→  1006→LEVEL 1 HIT: price reaches level1_target
   508→  1007→  1007→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   509→  1008→  1008→  book_qty    = ceil(qty × 0.50)                (book 50%)
   510→  1009→  1009→  remaining   = qty - book_qty
   511→  1010→  1010→  book_pnl    = book_qty × |level1_target - entry|
   512→  1011→  1011→  
   513→  1012→  1012→  Action:
   514→  1013→  1013→    → Exit book_qty shares at level1_target
   515→  1014→  1014→    → Move SL to entry price (risk-free now)
   516→  1015→  1015→    → Record partial booking in position.booked_levels
   517→  1016→  1016→    → Telegram: "Partial booking L1: {symbol} {book_qty} shares @ {level1_target}"
   518→  1017→  1017→    → WebSocket: push updated position to browser
   519→  1018→  1018→
   520→  1019→  1019→
   521→  1020→  1020→LEVEL 2 HIT: price reaches level2_target
   522→  1021→  1021→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   523→  1022→  1022→  book_qty    = ceil(remaining × 0.50)          (book 50% of remaining = 25% of original)
   524→  1023→  1023→  remaining   = remaining - book_qty
   525→  1024→  1024→  book_pnl    = book_qty × |level2_target - entry|
   526→  1025→  1025→  
   527→  1026→  1026→  Action:
   528→  1027→  1027→    → Exit book_qty shares at level2_target
   529→  1028→  1028→    → Trail SL to level1_target (locked L1 profit)
   530→  1029→  1029→    → Record partial booking
   531→  1030→  1030→    → Telegram: "Partial booking L2: {symbol} {book_qty} shares @ {level2_target}"
   532→  1031→  1031→
   533→  1032→  1032→
   534→  1033→  1033→LEVEL 3 HIT: price reaches level3_target (original target)
   535→  1034→  1034→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   536→  1035→  1035→  Option A: Exit remaining shares at target
   537→  1036→  1036→  Option B: Trail SL with Supertrend and let it run (if momentum is strong)
   538→  1037→  1037→  
   539→  1038→  1038→  Default: Option A (exit at target)
   540→  1039→  1039→  Config:  User can choose Option B per strategy
   541→  1040→  1040→
   542→  1041→  1041→
   543→  1042→  1042→TRAILING STOP LOSS
   544→  1043→  1043→━━━━━━━━━━━━━━━━━━━
   545→  1044→  1044→  After Level 1 booked (risk-free):
   546→  1045→  1045→    → Trail SL method: Step trail or Supertrend trail
   547→  1046→  1046→    
   548→  1047→  1047→    Step Trail:
   549→  1048→  1048→      → Every time price moves +1% from last SL, move SL up by 0.5%
   550→  1049→  1049→      → SL never moves down
   551→  1050→  1050→    
   552→  1051→  1051→    Supertrend Trail:
   553→  1052→  1052→      → Use Supertrend indicator as trailing SL
   554→  1053→  1053→      → More adaptive, follows price action
   555→  1054→  1054→```
