   811→   811→━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   812→   812→  Check:    How many positions already in same sector as new signal
   813→   813→  Default:  max_per_sector = 2
   814→   814→  Action:   REJECT if sector limit reached
   815→   815→  Why:      Don't put all money in banking if you already have 2 bank stocks
   816→   816→  Sectors:  Banking, IT, Energy, Pharma, Auto, FMCG, Infra, Metal, etc.
   817→   817→
   818→   818→
   819→   819→GATE 3: MAX POSITION SIZE
   820→   820→━━━━━━━━━━━━━━━━━━━━━━━━━━
   821→   821→  Check:    position_value > max_position_pct × total_capital
   822→   822→  Default:  max_position_pct = 25%
   823→   823→  Action:   REJECT if exceeds (or resize if possible)
   824→   824→  Also:     Check available capital (balance aware)
   825→   825→
   826→   826→
   827→   827→GATE 4: MAX DAILY TRADES
   828→   828→━━━━━━━━━━━━━━━━━━━━━━━━━━
   829→   829→  Check:    trades_today >= max_daily_trades
   830→   830→  Default:  max_daily_trades = 10
   831→   831→  Action:   REJECT new signals for rest of day
   832→   832→
   833→   833→
   834→   834→GATE 5: MAX DAILY LOSS
   835→   835→━━━━━━━━━━━━━━━━━━━━━━━━━━
   836→   836→  Check:    daily_pnl <= -max_daily_loss
   837→   837→  Default:  max_daily_loss = 3% of capital
   838→   838→  Action:   REJECT new trades + tighten existing SLs
   839→   839→  Warning:  At 67% usage → reduce new position sizes by 50%
   840→   840→
   841→   841→
   842→   842→GATE 6: CORRELATION CHECK
   843→   843→━━━━━━━━━━━━━━━━━━━━━━━━━━
   844→   844→  Check:    Is new signal highly correlated with existing positions?
   845→   845→  Default:  max_correlated = 2
   846→   846→  Action:   REJECT if too many correlated positions
   847→   847→  Example:  Already have SBI + HDFCBANK → reject PNB (all banking)

   849→   849→

   852→   852→  Check:    current_vix > vix_threshold
   853→   853→  Default:  vix_threshold = 20
   854→   854→  Action:   REJECT new trades when VIX too high
   855→   855→  Why:      High VIX = high volatility = unpredictable = dangerous

   857→   857→

   859→   859→━━━━━━━━━━━━━━━━━━━━━
   860→   860→  Check:    Current time outside preferred trading window
   861→   861→  Default:  New trades: 9:30 AM - 2:30 PM only
   862→   862→  Action:   REJECT new trades outside window
   863→   863→  Why:      First 15 min = ORB formation. Last hour = squaring off.
   864→   864→  Note:     Can configure different windows per strategy
   865→   865→
   866→   866→
   867→   867→GATE 9: PRICE MISMATCH GUARD
   868→   868→━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   869→   869→  Check:    |signal_entry_price - broker_live_ltp| / broker_live_ltp > threshold
   870→   870→  Default:  threshold = 0.5%
   871→   871→  Action:   REJECT + refresh feed + alert

   873→   873→  Critical: This is the P0 bug guard

   877→   877→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   878→   878→  Check:    kronos_score < min_confidence
   879→   879→  Default:  min_confidence = 0.6
   880→   880→  Action:   REJECT low-confidence signals
   881→   881→  Why:      Don't trade weak signals. Wait for strong ones.
   882→   882→
   883→   883→

   885→   885→━━━━━━━━━━━━━━━━━━━━━━━
   886→   886→  Check:    current_drawdown > max_drawdown_pct
   887→   887→  Default:  max_drawdown_pct = 5% (per day)
   888→   888→  Action:   PAUSE engine + alert

   890→   890→
   891→   891→
   892→   892→GATE 12: MARGIN CHECK

   894→   894→  Check:    required_margin > available_margin
   895→   895→  Action:   REJECT or RESIZE to fit available margin
   896→   896→  Live mode: Check actual broker margin
   897→   897→  Paper mode: Check virtual capital
   898→   898→
   899→   899→
   900→   900→GATE 13: DUPLICATE SIGNAL
   901→   901→━━━━━━━━━━━━━━━━━━━━━━━━━━
   902→   902→  Check:    Same symbol + same direction signal in last X minutes?
   903→   903→  Default:  X = 15 minutes
   904→   904→  Action:   REJECT duplicate (keep first one)
   905→   905→  Why:      Multiple strategies may fire same signal. Take only first.

   910→   910→## 3.5 Dynamic Position Sizing — Detailed Logic
   911→   911→
   912→   912→```
   913→   913→STEP 1: BASE KELLY FRACTION
   914→   914→━━━━━━━━━━━━━━━━━━━━━━━━━━━
   915→   915→  win_rate     = from strategy_performance table (historical)
   916→   916→  avg_win      = average winning trade pnl
   917→   917→  avg_loss     = average losing trade pnl
   918→   918→  kelly_f      = (win_rate × avg_win - (1 - win_rate) × avg_loss) / avg_win
   919→   919→  kelly_f      = clamp(kelly_f, 0.02, 0.25)  // Never below 2%, never above 25%
   920→   920→  
   921→   921→  base_size    = total_capital × kelly_f
   922→   922→
   923→   923→
   924→   924→STEP 2: CONFIDENCE ADJUSTMENT
   925→   925→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   926→   926→  kronos_score = signal.confidence  (0 to 1)
   927→   927→  
   928→   928→  if kronos_score >= 0.8:   confidence_mult = 1.0    (strong signal)
   929→   929→  if kronos_score >= 0.6:   confidence_mult = 0.8    (decent signal)
   930→   930→  if kronos_score >= 0.4:   confidence_mult = 0.5    (weak signal)
   931→   931→  if kronos_score <  0.4:   REJECT by G10           (too weak)
   932→   932→  
   933→   933→  adjusted_size = base_size × confidence_mult
   934→   934→
   935→   935→
   936→   936→STEP 3: VOLATILITY ADJUSTMENT
   937→   937→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   938→   938→  vix = current_vix
   939→   939→  
   940→   940→  if vix <= 14:   vol_mult = 1.0    (calm market, normal size)
   941→   941→  if vix <= 18:   vol_mult = 0.85   (slightly nervous)
   942→   942→  if vix <= 22:   vol_mult = 0.65   (worried, reduce)
   943→   943→  if vix >  22:   vol_mult = 0.4    (fearful, much smaller)
   944→   944→  
   945→   945→  adjusted_size = adjusted_size × vol_mult
   946→   946→
   947→   947→
   948→   948→STEP 4: DRAWDOWN ADJUSTMENT
   949→   949→━━━━━━━━━━━━━━━━━━━━━━━━━━━
   950→   950→  daily_pnl_pct = day_pnl / total_capital
   951→   951→  
   952→   952→  if daily_pnl_pct >= 0:      dd_mult = 1.0    (profitable day, normal)
   953→   953→  if daily_pnl_pct >= -1%:    dd_mult = 0.9    (small loss, slight reduce)
   954→   954→  if daily_pnl_pct >= -2%:    dd_mult = 0.7    (moderate loss, reduce)
   955→   955→  if daily_pnl_pct >= -3%:    dd_mult = 0.4    (approaching limit, heavy reduce)
   956→   956→  if daily_pnl_pct <  -3%:    STOP new trades  (limit hit)
   957→   957→  
   958→   958→  adjusted_size = adjusted_size × dd_mult
   959→   959→
   960→   960→
   961→   961→STEP 5: CAPITAL AVAILABILITY
   962→   962→━━━━━━━━━━━━━━━━━━━━━━━━━━━
   963→   963→  available = free_capital × 0.9   (never use 100%)
   964→   964→  
   965→   965→  if adjusted_size <= available:
   966→   966→      final_size = adjusted_size   (we can afford it)
   967→   967→  else:
   968→   968→      final_size = available       (cap at what we can afford)
   969→   969→  
   970→   970→  if final_size < min_position_size (e.g., ₹5,000):
   971→   971→      REJECT: "Position too small after adjustments"
   972→   972→
   973→   973→
   974→   974→STEP 6: CONVERT TO QUANTITY
   975→   975→━━━━━━━━━━━━━━━━━━━━━━━━━━━
   976→   976→  qty = floor(final_size / entry_price)
   977→   977→  
   978→   978→  if segment == "OPT":
   979→   979→      lot_size = option_lot_size[symbol]  (e.g., RELIANCE lot = 250)
   980→   980→      qty = floor(final_size / (premium × lot_size)) × lot_size
   981→   981→  
   982→   982→  final_capital_required = qty × entry_price  (or qty × premium × lot_size for options)
   983→   983→```

   987→   987→## 3.6 Partial Profit Booking — Detailed Logic
   988→   988→
   989→   989→```
   990→   990→WHEN POSITION IS OPENED:
   991→   991→━━━━━━━━━━━━━━━━━━━━━━━━
   992→   992→  entry  = position.entry_price
   993→   993→  sl     = position.sl_price
   994→   994→  target = position.target_price
   995→   995→  qty    = position.total_quantity
   996→   996→  
   997→   997→  risk   = |entry - sl|                          (risk per share)
   998→   998→  
   999→   999→  level1_target = entry + (risk × 1.0)          (1:1 R:R)
  1000→  1000→  level2_target = entry + (risk × 1.5)          (1.5:1 R:R)
  1001→  1001→  level3_target = entry + (risk × 2.0)          (2:1 R:R) = original target
  1002→  1002→  
  1003→  1003→  (For SELL positions, subtract instead of add)
  1004→  1004→
  1005→  1005→
  1006→  1006→LEVEL 1 HIT: price reaches level1_target
  1007→  1007→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  1008→  1008→  book_qty    = ceil(qty × 0.50)                (book 50%)
  1009→  1009→  remaining   = qty - book_qty
  1010→  1010→  book_pnl    = book_qty × |level1_target - entry|
  1011→  1011→  
  1012→  1012→  Action:
  1013→  1013→    → Exit book_qty shares at level1_target
  1014→  1014→    → Move SL to entry price (risk-free now)
  1015→  1015→    → Record partial booking in position.booked_levels
  1016→  1016→    → Telegram: "Partial booking L1: {symbol} {book_qty} shares @ {level1_target}"
  1017→  1017→    → WebSocket: push updated position to browser
  1018→  1018→
  1019→  1019→
  1020→  1020→LEVEL 2 HIT: price reaches level2_target
  1021→  1021→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  1022→  1022→  book_qty    = ceil(remaining × 0.50)          (book 50% of remaining = 25% of original)
  1023→  1023→  remaining   = remaining - book_qty
  1024→  1024→  book_pnl    = book_qty × |level2_target - entry|
  1025→  1025→  
  1026→  1026→  Action:
  1027→  1027→    → Exit book_qty shares at level2_target
  1028→  1028→    → Trail SL to level1_target (locked L1 profit)
  1029→  1029→    → Record partial booking
  1030→  1030→    → Telegram: "Partial booking L2: {symbol} {book_qty} shares @ {level2_target}"
  1031→  1031→
  1032→  1032→
  1033→  1033→LEVEL 3 HIT: price reaches level3_target (original target)
  1034→  1034→━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  1035→  1035→  Option A: Exit remaining shares at target
  1036→  1036→  Option B: Trail SL with Supertrend and let it run (if momentum is strong)
  1037→  1037→  
  1038→  1038→  Default: Option A (exit at target)
  1039→  1039→  Config:  User can choose Option B per strategy
  1040→  1040→
  1041→  1041→
  1042→  1042→TRAILING STOP LOSS
  1043→  1043→━━━━━━━━━━━━━━━━━━━
  1044→  1044→  After Level 1 booked (risk-free):
  1045→  1045→    → Trail SL method: Step trail or Supertrend trail
  1046→  1046→    
  1047→  1047→    Step Trail:
  1048→  1048→      → Every time price moves +1% from last SL, move SL up by 0.5%
  1049→  1049→      → SL never moves down
  1050→  1050→    
  1051→  1051→    Supertrend Trail:
  1052→  1052→      → Use Supertrend indicator as trailing SL
  1053→  1053→      → More adaptive, follows price action
  1054→  1054→```
