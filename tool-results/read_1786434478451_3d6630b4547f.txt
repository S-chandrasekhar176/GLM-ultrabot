   183→function formatINR(value: number): string {
   184→  const sign = value < 0 ? '-' : '';
   185→  const abs = Math.abs(value);
   186→  const parts = abs.toFixed(2).split('.');
   187→  const intPart = parts[0];
   188→  const decPart = parts[1];
   189→  let formatted = '';
   190→  if (intPart.length <= 3) {
   191→    formatted = intPart;
   192→  } else {
   193→    const last3 = intPart.slice(-3);
   194→    const rest = intPart.slice(0, -3);
   195→    formatted = rest.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',' + last3;
   196→  }
   197→  return `₹${sign}${formatted}.${decPart}`;
   198→}
   199→
   200→function formatPercent(value: number): string {
   201→  const sign = value >= 0 ? '+' : '';
   202→  return `${sign}${value.toFixed(2)}%`;
   203→}

   209→function CircularProgress({ value, size = 64, strokeWidth = 5, color = '#00d09c' }: {
   210→  value: number;
   211→  size?: number;
   212→  strokeWidth?: number;
   213→  color?: string;
   214→}) {
   215→  const radius = (size - strokeWidth) / 2;
   216→  const circumference = 2 * Math.PI * radius;
   217→  const offset = circumference - (value / 100) * circumference;
   218→
   219→  return (
   220→    <svg width={size} height={size} className="-rotate-90">
   221→      <circle
   222→        cx={size / 2}
   223→        cy={size / 2}
   224→        r={radius}
   225→        fill="none"
   226→        stroke="#1e293b"
   227→        strokeWidth={strokeWidth}
   228→      />
   229→      <circle
   230→        cx={size / 2}
   231→        cy={size / 2}
   232→        r={radius}
   233→        fill="none"
   234→        stroke={color}
   235→        strokeWidth={strokeWidth}
   236→        strokeDasharray={circumference}
   237→        strokeDashoffset={offset}
   238→        strokeLinecap="round"
   239→        className="transition-all duration-700 ease-out"
   240→      />
   241→      <text
   242→        x={size / 2}
   243→        y={size / 2}
   244→        className="fill-ub-text-primary text-xs font-bold"
   245→        textAnchor="middle"
   246→        dominantBaseline="central"
   247→        transform={`rotate(90, ${size / 2}, ${size / 2})`}
   248→      >
   249→        {value.toFixed(0)}%
   250→      </text>
   251→    </svg>
   252→  );
   253→}

   306→function UptimeTicker({ startedAt }: { startedAt: number | null }) {
   307→  const [uptime, setUptime] = useState('00:00:00');
   308→
   309→  useEffect(() => {
   310→    if (!startedAt) { setUptime('00:00:00'); return; }
   311→    const tick = () => {
   312→      const diff = Math.floor((Date.now() - startedAt) / 1000);
   313→      const h = String(Math.floor(diff / 3600)).padStart(2, '0');
   314→      const m = String(Math.floor((diff % 3600) / 60)).padStart(2, '0');
   315→      const s = String(diff % 60).padStart(2, '0');
   316→      setUptime(`${h}:${m}:${s}`);
   317→    };
   318→    tick();
   319→    const interval = setInterval(tick, 1000);
   320→    return () => clearInterval(interval);
   321→  }, [startedAt]);
   322→
   323→  return (
   324→    <span className="font-mono text-xs" style={{ color: '#00D09C' }}>
   325→      {uptime}
   326→    </span>
   327→  );
   328→}

   389→function MarketTimer() {
   390→  const [now, setNow] = useState(() => new Date());
   391→  const engine = useEngineStore();

   402→    minute: '2-digit',
   403→    second: '2-digit',
   404→  });
   405→
   406→  const marketCloseSeconds = engine.marketCloseSeconds;
   407→  const hours = Math.floor(marketCloseSeconds / 3600);
   408→  const minutes = Math.floor((marketCloseSeconds % 3600) / 60);
   409→  const seconds = Math.floor(marketCloseSeconds % 60);
   410→  const pad = (n: number) => String(n).padStart(2, '0');
   411→  const timeToClose = marketCloseSeconds > 0
   412→    ? `${pad(hours)}:${pad(minutes)}:${pad(seconds)}`

   415→  const isUrgent = marketCloseSeconds > 0 && marketCloseSeconds < 1800;
   416→
   417→  return (

   419→      <div className="flex items-center gap-2">
   420→        <Clock className="h-4 w-4 text-ub-text-muted" />

   432→          {timeToClose}
