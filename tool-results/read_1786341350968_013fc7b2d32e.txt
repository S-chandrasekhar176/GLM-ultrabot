    97→/** Indian number formatting: ₹1,23,456.78 */
    98→function formatINR(amount: number, showSign = false): string {
    99→  const sign = showSign && amount > 0 ? '+' : '';
   100→  const formatted = new Intl.NumberFormat('en-IN', {
   101→    minimumFractionDigits: 2,
   102→    maximumFractionDigits: 2,
   103→  }).format(Math.abs(amount));
   104→  return `${sign}₹${formatted}`;
   105→}
   106→
   107→function formatINRCompact(amount: number, showSign = false): string {
   108→  const sign = showSign && amount > 0 ? '+' : '';
   109→  const formatted = new Intl.NumberFormat('en-IN', {
   110→    minimumFractionDigits: 0,
   111→    maximumFractionDigits: 0,
   112→  }).format(Math.abs(amount));
   113→  return `${sign}₹${formatted}`;
   114→}

   142→function CircularProgress({ value, size = 48, strokeWidth = 4 }: { value: number; size?: number; strokeWidth?: number }) {
   143→  const radius = (size - strokeWidth) / 2;
   144→  const circumference = 2 * Math.PI * radius;
   145→  const offset = circumference - (value / 100) * circumference;
   146→
   147→  return (
   148→    <svg width={size} height={size} className="-rotate-90">
   149→      <circle
   150→        cx={size / 2}
   151→        cy={size / 2}
   152→        r={radius}
   153→        fill="none"
   154→        stroke={theme.colors.surfaceActive}
   155→        strokeWidth={strokeWidth}
   156→      />
   157→      <circle
   158→        cx={size / 2}
   159→        cy={size / 2}
   160→        r={radius}
   161→        fill="none"
   162→        stroke={value >= 50 ? theme.colors.profit : theme.colors.warning}
   163→        strokeWidth={strokeWidth}
   164→        strokeDasharray={circumference}
   165→        strokeDashoffset={offset}
   166→        strokeLinecap="round"
   167→        className="transition-all duration-700 ease-out"
   168→      />
   169→    </svg>
   170→  );
   171→}

   398→function CapitalPanel({ data }: { data: DashboardData }) {
   399→  const total = data.totalCapital ?? 100000;
   400→  const used = data.capitalUsed ?? 65000;
   401→  const maxCapital = data.maxCapital ?? 90000;
   402→  const free = total - used;
   403→  const dayPnl = data.dayPnlAmount ?? 2340;
   404→  const dayPnlPct = total > 0 ? (dayPnl / total) * 100 : 0;
   405→  const inceptionPnl = data.totalPnlInception ?? 18420;
   406→  const usedPercent = maxCapital > 0 ? (used / maxCapital) * 100 : 0;
   407→
   408→  const rows = [
   409→    {
   410→      label: 'Total Capital',
   411→      value: formatINRCompact(total),
   412→      color: theme.colors.textPrimary,
   413→    },
   414→    {
   415→      label: 'Capital Used',
   416→      value: `${formatINRCompact(used)} / ${formatINRCompact(maxCapital)}`,
   417→      color: theme.colors.textPrimary,
   418→      progress: usedPercent,
   419→    },
   420→    {
   421→      label: 'Free Capital',
   422→      value: formatINRCompact(free),
   423→      color: theme.colors.accent,
   424→    },
   425→    {
   426→      label: 'Day P&L',
   427→      value: `${formatINR(dayPnl, true)} / ${pnlSign(dayPnlPct)}${dayPnlPct.toFixed(2)}%`,
   428→      color: pnlColor(dayPnl),
   429→    },
   430→    {
   431→      label: 'Total P&L (Inception)',
   432→      value: formatINR(inceptionPnl, true),
   433→      color: pnlColor(inceptionPnl),
   434→    },
   435→  ];
   436→
   437→  return (
   438→    <PanelCard>
   439→      <PanelHeader icon={BarChart3} title="Capital Overview" />
   440→      <CardContent className="px-4 pb-4 pt-0">
   441→        <div className="space-y-0">
   442→          {rows.map((row, i) => (
   443→            <div key={row.label}>
   444→              <div className="flex items-center justify-between py-2.5">
   445→                <span className="text-xs" style={{ color: theme.colors.textMuted }}>
   446→                  {row.label}
   447→                </span>
   448→                <span
   449→                  className="text-sm font-medium tabular-nums"
   450→                  style={{ color: row.color }}
   451→                >
   452→                  {row.value}
   453→                </span>
   454→              </div>
   455→              {row.progress !== undefined && (
   456→                <div className="-mt-1 mb-1">
   457→                  <Progress
   458→                    value={row.progress}
   459→                    className="h-1.5"
   460→                    style={{
   461→                      ['--progress-color' as string]: theme.colors.accent,
   462→                    }}
   463→                  />
   464→                </div>
   465→              )}
   466→              {i < rows.length - 1 && <Separator className="opacity-50" />}
   467→            </div>
   468→          ))}
   469→        </div>
   470→      </CardContent>
   471→    </PanelCard>
   472→  );
   473→}

   901→// Section 2 — Market Timer Card
   902→// ─────────────────────────────────────────────
   903→
   904→function MarketTimerCard() {
   905→  const [now, setNow] = useState(new Date());
   906→
   907→  useEffect(() => {
   908→    const interval = setInterval(() => setNow(new Date()), 1000);
   909→    return () => clearInterval(interval);
   910→  }, []);
   911→
   912→  // NSE closes at 15:30 IST
   913→  const istHours = now.getUTCHours() + 5; // Approximate IST (ignoring 30min for simplicity in display)
   914→  const istMinutes = now.getUTCMinutes() + 30;
   915→  const totalIstMinutes = istHours * 60 + istMinutes;
   916→  const closeMinutes = 15 * 60 + 30; // 15:30
   917→  const openMinutes = 9 * 60 + 15; // 09:15
   918→
   919→  let timeToClose = closeMinutes - totalIstMinutes;
   920→  let marketStatus: 'open' | 'closed' | 'pre-market' = 'closed';
   921→
   922→  if (totalIstMinutes >= openMinutes && totalIstMinutes <= closeMinutes) {
   923→    marketStatus = 'open';
   924→  } else if (totalIstMinutes < openMinutes) {
   925→    marketStatus = 'pre-market';
   926→    timeToClose = openMinutes - totalIstMinutes;
   927→  }
   928→
   929→  const hours = Math.floor(Math.max(timeToClose, 0) / 60);
   930→  const minutes = Math.max(timeToClose, 0) % 60;
   931→
   932→  const istTimeString = now.toLocaleTimeString('en-IN', {
   933→    timeZone: 'Asia/Kolkata',
   934→    hour: '2-digit',
   935→    minute: '2-digit',
   936→    second: '2-digit',
   937→    hour12: false,
   938→  });
   939→
   940→  return (
   941→    <PanelCard>
   942→      <CardContent className="p-4">
   943→        <div className="flex items-center gap-3">
   944→          <div
   945→            className="flex items-center justify-center rounded-lg p-2.5 shrink-0"
   946→            style={{ backgroundColor: `${theme.colors.accent}15` }}
   947→          >
   948→            <Clock size={18} style={{ color: theme.colors.accent }} />
   949→          </div>
   950→          <div className="flex-1 min-w-0">
   951→            <p className="text-xs" style={{ color: theme.colors.textMuted }}>
   952→              {marketStatus === 'open'
   953→                ? 'Market Closes In'
   954→                : marketStatus === 'pre-market'
   955→                  ? 'Market Opens In'
   956→                  : 'Market Closed'}
   957→            </p>
   958→            <p
   959→              className="text-lg font-bold tabular-nums"
   960→              style={{ color: theme.colors.textPrimary }}
   961→            >
   962→              {marketStatus === 'closed'
   963→                ? '—'
   964→                : `${hours}h ${String(minutes).padStart(2, '0')}m`}
   965→            </p>
   966→          </div>
   967→          <div className="text-right">
   968→            <p className="text-[10px]" style={{ color: theme.colors.textDisabled }}>
   969→              IST
   970→            </p>
   971→            <p
   972→              className="text-xs font-medium tabular-nums"
   973→              style={{ color: theme.colors.textMuted }}
   974→            >
   975→              {istTimeString}
   976→            </p>
   977→          </div>
   978→        </div>
   979→      </CardContent>
   980→    </PanelCard>
   981→  );
   982→}
