   302→function TimerCountdown({ expiryAt }: { expiryAt: string }) {
   303→  const [timeLeft, setTimeLeft] = useState('');
   304→  const [isExpired, setIsExpired] = useState(false);
   305→
   306→  useEffect(() => {
   307→    const update = () => {
   308→      const diff = new Date(expiryAt).getTime() - Date.now();
   309→      if (diff <= 0) {
   310→        setIsExpired(true);
   311→        setTimeLeft('Expired');
   312→        return;
   313→      }
   314→      const mins = Math.floor(diff / 60_000);
   315→      const secs = Math.floor((diff % 60_000) / 1000);
   316→      setTimeLeft(`${mins}:${secs.toString().padStart(2, '0')}`);
   317→    };
   318→    update();
   319→    const interval = setInterval(update, 1000);
   320→    return () => clearInterval(interval);
   321→  }, [expiryAt]);
   322→
   323→  return (
   324→    <span
   325→      className={`flex items-center gap-1 text-xs font-mono font-medium ${
   326→        isExpired ? 'text-ub-loss' : timeLeft.includes('0:') || timeLeft.includes('1:') ? 'text-ub-warning' : 'text-ub-text-muted'
   327→      }`}
   328→    >
   329→      <Timer className="h-3 w-3" />
   330→      {timeLeft}
   331→    </span>
   332→  );
   333→}

   339→function RiskGatesPanel({ gates }: { gates: RiskGate[] }) {

   352→          <span className="text-ub-border">|</span>

   855→type FilterTab = 'all' | 'pending' | 'confirmed' | 'skipped' | 'expired';
   856→
   857→export default function OpportunitiesPage() {
   858→  const [activeTab, setActiveTab] = useState<FilterTab>('all');
   859→  const [opportunities, setOpportunities] = useState<OpportunityData[]>(MOCK_OPPORTUNITIES);
   860→  const [isLoading, setIsLoading] = useState(false);
   861→  const [isConfirming, setIsConfirming] = useState(false);
   862→  const [isSkipping, setIsSkipping] = useState(false);
   863→  const [backtestLoading, setBacktestLoading] = useState<Record<string, boolean>>({});
   864→  const [backtestResults, setBacktestResults] = useState<Record<string, any>>({});
   865→  const backtestPollRef = useRef<Record<string, NodeJS.Timeout>>({});
   866→
   867→  // Load opportunities on mount
   868→  const loadOpportunities = useCallback(async () => {
   869→    setIsLoading(true);
   870→    try {
   871→      const data = await getOpportunities();
   872→      if (Array.isArray(data) && data.length > 0) {
   873→        setOpportunities(data.map(transformOpportunity));
   874→      }
   875→      // If backend returns empty or fails, keep MOCK_OPPORTUNITIES as fallback
   876→    } catch {
   877→      // Backend not available — keep mock data
   878→    } finally {
   879→      setIsLoading(false);
   880→    }
   881→  }, []);
   882→
   883→  useEffect(() => {
   884→    loadOpportunities();
   885→  }, [loadOpportunities]);
   886→
   887→  // Clean up polling timers on unmount
   888→  useEffect(() => {
   889→    return () => {
   890→      Object.values(backtestPollRef.current).forEach(clearTimeout);
   891→    };
   892→  }, []);
   893→
   894→  const filtered = useMemo(() => {
   895→    if (activeTab === 'all') return opportunities;
   896→    return opportunities.filter((o) => o.status === activeTab);
   897→  }, [opportunities, activeTab]);
   898→
   899→  const pendingCount = opportunities.filter((o) => o.status === 'pending').length;
   900→
   901→  const handleConfirm = useCallback(async (id: string) => {
   902→    // Optimistic update
   903→    setOpportunities(prev =>
   904→      prev.map(o => o.id === id ? { ...o, status: 'confirmed' as OppStatus } : o),
   905→    );
   906→    try {
   907→      await confirmOpportunity(id);
   908→      toast.success('Opportunity confirmed — engine will execute');
   909→    } catch (err: any) {
   910→      // Revert on failure
   911→      setOpportunities(prev =>
   912→        prev.map(o => o.id === id ? { ...o, status: 'pending' as OppStatus } : o),
   913→      );
   914→      toast.error(err?.response?.data?.detail || 'Failed to confirm opportunity');
   915→    }
   916→  }, []);
   917→
   918→  const handleSkip = useCallback(async (id: string) => {
   919→    setOpportunities(prev =>
   920→      prev.map(o => o.id === id ? { ...o, status: 'skipped' as OppStatus } : o),
   921→    );
   922→    try {
   923→      await skipOpportunity(id);
   924→      toast.success('Opportunity skipped');
   925→    } catch (err: any) {
   926→      setOpportunities(prev =>
   927→        prev.map(o => o.id === id ? { ...o, status: 'pending' as OppStatus } : o),
   928→      );
   929→      toast.error(err?.response?.data?.detail || 'Failed to skip opportunity');
   930→    }
   931→  }, []);
   932→
   933→  // Quick backtest: run a 30-day backtest for the opportunity's strategy + symbol
   934→  const handleQuickBacktest = useCallback(async (id: string) => {
   935→    const opp = opportunities.find(o => o.id === id);
   936→    if (!opp) return;
   937→
   938→    setBacktestLoading(prev => ({ ...prev, [id]: true }));
   939→    setBacktestResults(prev => { const next = { ...prev }; delete next[id]; return next; });
   940→
   941→    // Cancel any existing poll for this id
   942→    if (backtestPollRef.current[id]) clearTimeout(backtestPollRef.current[id]);
   943→
   944→    try {
   945→      const endDate = new Date().toISOString().split('T')[0];
   946→      const startDate = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString().split('T')[0];
   947→      const strategyKey = opp.strategy.toLowerCase().replace(/ /g, '_');
   948→
   949→      const response: any = await runBacktest({
   950→        strategy: strategyKey,
   951→        symbol: opp.symbol,
   952→        start_date: startDate,
   953→        end_date: endDate,
   954→        timeframe: '5min',
   955→        initial_capital: 100000,
   956→      });
   957→
   958→      if (response?.run_id) {
   959→        const pollStatus = async () => {
   960→          try {
   961→            const status: any = await getBacktestStatus(response.run_id);
   962→            if (status?.status === 'completed') {
   963→              const result: any = await getBacktestResult(response.run_id);
   964→              setBacktestResults(prev => ({
   965→                ...prev,
   966→                [id]: {
   967→                  winRate: (status.win_rate ?? result?.win_rate ?? 0) * 100,
