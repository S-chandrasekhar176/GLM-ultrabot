   180→   180→   180→
   181→   181→   181→function formatINR(value: number): string {
   182→   182→   182→  const sign = value < 0 ? '-' : '';
   183→   183→   183→  const abs = Math.abs(value);
   184→   184→   184→  const parts = abs.toFixed(2).split('.');
   185→   185→   185→  const intPart = parts[0];
   186→   186→   186→  const decPart = parts[1];
   187→   187→   187→  let formatted = '';
   188→   188→   188→  if (intPart.length <= 3) {
   189→   189→   189→    formatted = intPart;
   190→   190→   190→  } else {
   191→   191→   191→    const last3 = intPart.slice(-3);
   192→   192→   192→    const rest = intPart.slice(0, -3);
   193→   193→   193→    formatted = rest.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',' + last3;
   194→   194→   194→  }
   195→   195→   195→  return `₹${sign}${formatted}.${decPart}`;
   196→   196→   196→}

   198→   198→   198→function formatPercent(value: number): string {
   199→   199→   199→  const sign = value >= 0 ? '+' : '';
   200→   200→   200→  return `${sign}${value.toFixed(2)}%`;
   201→   201→   201→}
