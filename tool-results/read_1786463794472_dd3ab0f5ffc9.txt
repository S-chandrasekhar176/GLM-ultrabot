   484→// ─────────────────────────────────────────────
   485→// Positions Tab Component
   486→// ─────────────────────────────────────────────
   487→
   488→function PositionsTab({ resetSignal }: { resetSignal: number }) {
   489→  const [positions, setPositions] = useState<Position[]>(MOCK_POSITIONS);
   490→  const [closeDialogId, setCloseDialogId] = useState<string | null>(null);
   491→  const [modifyDialog, setModifyDialog] = useState<{
   492→    open: boolean;
   493→    id: string;
   494→    field: 'stopLoss' | 'target';
   495→  }>({ open: false, id: '', field: 'stopLoss' });
   496→  const [modifyValue, setModifyValue] = useState('');
   497→  const [partialDialog, setPartialDialog] = useState<{
   498→    open: boolean;
   499→    id: string;
   500→  }>({ open: false, id: '' });
   501→  const [partialQty, setPartialQty] = useState('');
   502→
   503→  // Reset paper mode when parent triggers
   504→  useEffect(() => {
   505→    if (resetSignal > 0) setPositions([]);
   506→  }, [resetSignal]);
   507→
   508→  const totalUnrealizedPnl = useMemo(
   509→    () => positions.reduce((sum, p) => sum + p.unrealizedPnl, 0),
   510→    [positions]
   511→  );
   512→
   513→  const totalInvested = useMemo(
   514→    () => positions.reduce((sum, p) => sum + p.entry * p.quantity, 0),
   515→    [positions]
   516→  );
   517→
   518→  const handleClosePosition = (id: string) => {
   519→    setPositions((prev) => prev.filter((p) => p.id !== id));
   520→    setCloseDialogId(null);
   521→    const pos = positions.find((p) => p.id === id);
   522→    toast.success(`${pos?.symbol} position closed`, {
   523→      description: `P&L: ${INR(pos?.unrealizedPnl ?? 0)}`,
   524→    });
   525→  };
   526→
   527→  const handleModify = () => {
   528→    const val = parseFloat(modifyValue);
   529→    if (isNaN(val) || val <= 0) return;
   530→    setPositions((prev) =>
   531→      prev.map((p) =>
   532→        p.id === modifyDialog.id
   533→          ? { ...p, [modifyDialog.field]: val }
   534→          : p
   535→      )
   536→    );
   537→    const pos = positions.find((p) => p.id === modifyDialog.id);
   538→    toast.success(`${pos?.symbol} ${modifyDialog.field === 'stopLoss' ? 'SL' : 'Target'} updated`, {
   539→      description: `New ${modifyDialog.field === 'stopLoss' ? 'Stop Loss' : 'Target'}: ${INR(val)}`,
   540→    });
   541→    setModifyDialog({ open: false, id: '', field: 'stopLoss' });
   542→    setModifyValue('');
   543→  };
   544→
   545→  const handlePartialClose = () => {
   546→    const qty = parseInt(partialQty, 10);
   547→    if (isNaN(qty) || qty <= 0) return;
   548→    setPositions((prev) =>
   549→      prev.map((p) =>
   550→        p.id === partialDialog.id
   551→          ? {
   552→              ...p,
   553→              remainingQty: Math.max(0, p.remainingQty - qty),
   554→              quantity: p.quantity,
   555→            }
   556→          : p
   557→      )
   558→    );
   559→    const pos = positions.find((p) => p.id === partialDialog.id);
   560→    toast.success(`Partial close on ${pos?.symbol}`, {
   561→      description: `${qty} shares closed, ${Math.max(0, (pos?.remainingQty ?? 0) - qty)} remaining.`,
   562→    });
   563→    setPartialDialog({ open: false, id: '' });
   564→    setPartialQty('');
   565→  };
   566→
   567→  const openModify = (id: string, field: 'stopLoss' | 'target') => {
   568→    const pos = positions.find((p) => p.id === id);
   569→ setModifyValue(pos ? String(pos[field]) : '');
   570→    setModifyDialog({ open: true, id, field });
   571→  };
   572→
   573→  const openPartial = (id: string) => {
   574→    const pos = positions.find((p) => p.id === id);
   575→    setPartialQty(pos ? String(pos.remainingQty) : '');
   576→    setPartialDialog({ open: true, id });
   577→  };
   578→
   579→  return (
   580→    <div className="space-y-4">
   581→      {/* Summary Bar */}
   582→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   583→        <Card className="bg-ub-surface border-ub-border">
   584→          <CardContent className="p-4 flex items-center gap-3">
   585→            <div className="h-9 w-9 rounded-lg bg-ub-accent/10 flex items-center justify-center">
   586→              <ArrowLeftRight className="h-4.5 w-4.5 text-ub-accent" />
   587→            </div>
   588→            <div>
   589→              <p className="text-[11px] uppercase tracking-wider text-ub-text-muted">Total Positions</p>
   590→              <p className="text-xl font-bold text-ub-text-primary">{positions.length}</p>
   591→            </div>
   592→          </CardContent>
   593→        </Card>
   594→        <Card className="bg-ub-surface border-ub-border">
   595→          <CardContent className="p-4 flex items-center gap-3">
   596→            <div
   597→              className={`h-9 w-9 rounded-lg flex items-center justify-center ${
   598→                totalUnrealizedPnl >= 0 ? 'bg-ub-profit/10' : 'bg-ub-loss/10'
   599→              }`}
   600→            >
   601→              {totalUnrealizedPnl >= 0 ? (
   602→                <TrendingUp className="h-4.5 w-4.5 text-ub-profit" />
   603→              ) : (
   604→                <TrendingDown className="h-4.5 w-4.5 text-ub-loss" />
   605→              )}
   606→            </div>
   607→            <div>
   608→              <p className="text-[11px] uppercase tracking-wider text-ub-text-muted">Unrealized P&L</p>
   609→              <p
   610→                className={`text-xl font-bold ${
   611→                  totalUnrealizedPnl >= 0 ? 'text-ub-profit' : 'text-ub-loss'
   612→                }`}
   613→              >
   614→                {INR(totalUnrealizedPnl)}
   615→              </p>
   616→            </div>
   617→          </CardContent>
   618→        </Card>
   619→        <Card className="bg-ub-surface border-ub-border">
   620→          <CardContent className="p-4 flex items-center gap-3">
   621→            <div className="h-9 w-9 rounded-lg bg-ub-warning/10 flex items-center justify-center">
   622→              <Target className="h-4.5 w-4.5 text-ub-warning" />
